BeamAngleDialog::BeamAngleDialog(QWidget *parent, double initialAngle)
    : QDialog(parent)
{
    setWindowFlags(Qt::Dialog | Qt::FramelessWindowHint);
    setFixedSize(450, 280);

    setStyleSheet(R"(
        QDialog {
            background-color: #202124;
            border: 1px solid #3c4043;
            border-radius: 15px;
        }

        QLabel {
            color: white;
            font-size: 15px;
        }

        QDoubleSpinBox {
            background-color: #303134;
            color: white;
            border: 1px solid #5f6368;
            border-radius: 6px;
            padding: 5px;
        }

        QPushButton {
            background-color: #3c4043;
            color: white;
            border: none;
            border-radius: 8px;
            padding: 9px 20px;
            font-size: 14px;
        }

        QPushButton:hover {
            background-color: #505357;
        }

        QPushButton:pressed {
            background-color: #303134;
        }
    )");

    titleLabel = new QLabel("Направление луча", this);
    titleLabel->setAlignment(Qt::AlignCenter);
    titleLabel->setStyleSheet("font-size: 20px; font-weight: bold;");

    hintLabel = new QLabel(
        QString("Угол между стеной и лучом (в сторону комнаты), от %1° до %2° включительно. "
                "90° - перпендикулярно стене. "
                "Для кривой стены берётся касательная в точке луча.")
            .arg(BeamAngle::Min, 0, 'f', BeamAngle::Decimals)
            .arg(BeamAngle::Max, 0, 'f', BeamAngle::Decimals),
        this
        );
    hintLabel->setAlignment(Qt::AlignCenter);
    hintLabel->setWordWrap(true);

    QLabel *angleLabel = new QLabel("Угол:", this);

    angleSpinBox = new QDoubleSpinBox(this);
    angleSpinBox->setDecimals(BeamAngle::Decimals);   // сначала знаки, потом диапазон
    angleSpinBox->setRange(BeamAngle::Min, BeamAngle::Max);
    angleSpinBox->setSuffix("°");
    angleSpinBox->setValue(initialAngle);

    // Enter в поле обрабатываем сами (см. eventFilter): иначе спинбокс молча
    // подменит недопустимый ввод прежним значением
    // (lineEdit() у QAbstractSpinBox защищён, поэтому ищем поле как дочерний виджет)
    angleEdit = angleSpinBox->findChild<QLineEdit *>();
    if (angleEdit)
        angleEdit->installEventFilter(this);

    QHBoxLayout *angleLayout = new QHBoxLayout;
    angleLayout->addWidget(angleLabel);
    angleLayout->addWidget(angleSpinBox);

    cancelButton = new QPushButton("Отмена", this);
    okButton = new QPushButton("Задать", this);

    // Кнопки не берут фокус: иначе при клике спинбокс теряет фокус и успевает
    // молча заменить недопустимый ввод (например, 0) прежним значением
    cancelButton->setFocusPolicy(Qt::NoFocus);
    okButton->setFocusPolicy(Qt::NoFocus);
    cancelButton->setAutoDefault(false);
    okButton->setAutoDefault(false);

    QHBoxLayout *buttonsLayout = new QHBoxLayout;
    buttonsLayout->addStretch();
    buttonsLayout->addWidget(cancelButton);
    buttonsLayout->addWidget(okButton);
    buttonsLayout->addStretch();

    QVBoxLayout *mainLayout = new QVBoxLayout(this);
    mainLayout->setContentsMargins(30, 25, 30, 25);
    mainLayout->setSpacing(15);

    mainLayout->addWidget(titleLabel);
    mainLayout->addWidget(hintLabel);
    mainLayout->addLayout(angleLayout);
    mainLayout->addStretch();
    mainLayout->addLayout(buttonsLayout);

    connect(cancelButton, &QPushButton::clicked,
            this, &QDialog::reject);

    connect(okButton, &QPushButton::clicked,
            this, &BeamAngleDialog::applyChanges);
}

bool BeamAngleDialog::eventFilter(QObject *obj, QEvent *event)
{
    if (angleEdit && obj == angleEdit && event->type() == QEvent::KeyPress)
    {
        const int key = static_cast<QKeyEvent *>(event)->key();
        if (key == Qt::Key_Return || key == Qt::Key_Enter)
        {
            applyChanges();
            return true;
        }
    }
    return QDialog::eventFilter(obj, event);
}

// Проверяет именно набранный текст (а не последнее "принятое" значение спинбокса).
// Недопустимый угол: диалог остаётся открытым, сообщение об ошибке уходит в чат.
void BeamAngleDialog::applyChanges()
{
    const QString typed = angleSpinBox->cleanText().trimmed();

    QString normalized = typed;
    normalized.replace(',', '.');

    bool ok = false;
    const double value = normalized.toDouble(&ok);

    if (!ok || value < BeamAngle::Min || value > BeamAngle::Max)
    {
        emit invalidAngle(
            QString("Угол должен быть числом от %1° до %2° включительно (введено «%3»). "
                    "Значение не принято: исправьте угол.")
                .arg(BeamAngle::Min, 0, 'f', BeamAngle::Decimals)
                .arg(BeamAngle::Max, 0, 'f', BeamAngle::Decimals)
                .arg(typed.isEmpty() ? QString("пусто") : typed));
        return;
    }

    angle_ = value;
    accept();
}

double BeamAngleDialog::getAngle() const
{
    return angle_;
}

    ySpinBox->setValue(point.getY());

    QHBoxLayout *coordinatesLayout = new QHBoxLayout;

    coordinatesLayout->addWidget(xLabel);
    coordinatesLayout->addWidget(xSpinBox);
    coordinatesLayout->addWidget(yLabel);
    coordinatesLayout->addWidget(ySpinBox);

    QLabel *mirrorLabel = new QLabel(
        "Выберите тип зеркала, исходящий из этой точки",
        this
        );

    mirrorLabel->setAlignment(Qt::AlignCenter);
    mirrorLabel->setWordWrap(true);

    mirrorType = new QComboBox(this);

    mirrorType->addItem("Плоское зеркало");
    mirrorType->addItem("Вогнутое зеркало");
    mirrorType->addItem("Выпуклое зеркало");

    QLabel *radiusLabel = new QLabel(
        "Радиус кривизны:",
        this
        );

    radiusSpinBox = new QDoubleSpinBox(this);

    radiusSpinBox->setRange(0.000001, 1000000.0);
    radiusSpinBox->setDecimals(6);
    radiusSpinBox->setEnabled(false);

    QHBoxLayout *radiusLayout = new QHBoxLayout;

    radiusLayout->addWidget(radiusLabel);
    radiusLayout->addWidget(radiusSpinBox);

    cancelButton = new QPushButton("Отмена", this);
    changeButton = new QPushButton("Изменить", this);

    QHBoxLayout *buttonsLayout = new QHBoxLayout;

    buttonsLayout->addStretch();
    buttonsLayout->addWidget(cancelButton);
    buttonsLayout->addWidget(changeButton);
    buttonsLayout->addStretch();

    QVBoxLayout *mainLayout = new QVBoxLayout(this);

    mainLayout->setContentsMargins(30, 25, 30, 25);
    mainLayout->setSpacing(15);

    mainLayout->addWidget(pointLabel);
    mainLayout->addLayout(coordinatesLayout);
    mainLayout->addWidget(mirrorLabel);
    mainLayout->addWidget(mirrorType);
    mainLayout->addLayout(radiusLayout);
    mainLayout->addStretch();
    mainLayout->addLayout(buttonsLayout);

    // Подгружаем текущий тип зеркала и радиус (для последней точки зеркала нет)
    if (mirror_ && !lastPoint)
    {
        int kind = MirrorKind::Plane;
        if (dynamic_cast<const ConcaveMirror *>(mirror_))
            kind = MirrorKind::Concave;
        else if (dynamic_cast<const ConvexMirror *>(mirror_))
            kind = MirrorKind::Convex;

        mirrorType->setCurrentIndex(kind);   // порядок списка совпадает с MirrorKind

        if (kind != MirrorKind::Plane)
            radiusSpinBox->setValue(mirror_->getRadius());   // модуль радиуса

        radiusSpinBox->setEnabled(kind != MirrorKind::Plane);
    }

    // Для последней точки незамкнутого полигона
    // зеркало из неё ещё не существует.
    if (lastPoint)
    {
        mirrorType->setEnabled(false);
        radiusSpinBox->setEnabled(false);
    }

    connect(
        mirrorType,
        &QComboBox::currentIndexChanged,
        this,
        &ChangePointDialog::changeMirrorType
        );

    connect(
        cancelButton,
        &QPushButton::clicked,
        this,
        &QDialog::reject
        );

    connect(
        changeButton,
        &QPushButton::clicked,
        this,
        &ChangePointDialog::applyChanges
        );
}

void ChangePointDialog::changeMirrorType()
{
    radiusSpinBox->setEnabled(mirrorType->currentIndex() != 0);
}

void ChangePointDialog::applyChanges()
{
    data_.x = xSpinBox->value();
    data_.y = ySpinBox->value();
    data_.mirrorType = mirrorType->currentIndex();
    data_.radius = radiusSpinBox->value();

    accept();
}

ChangePointData ChangePointDialog::getData() const
{
    return data_;
}

// Тип зеркала (см. MirrorKind)
int TypeOf(const Mirror &m)
{
    if (dynamic_cast<const ConcaveMirror *>(&m)) return MirrorKind::Concave;
    if (dynamic_cast<const ConvexMirror *>(&m))  return MirrorKind::Convex;
    return MirrorKind::Plane;
}

// Создаёт зеркало нужного типа. Если радиус меньше половины длины зеркала,
// кривое зеркало невозможно, поэтому вместо него делается плоское
// (*flattened = true), а исключение не бросается.
std::unique_ptr<Mirror> MakeMirror(int type,
                                   const Points &a,
                                   const Points &b,
                                   double radius,
                                   bool *flattened)
{
    if (flattened) *flattened = false;

    if (type != MirrorKind::Plane && !(radius >= a.Dist(b) / 2.0)) {
        if (flattened) *flattened = true;
        type = MirrorKind::Plane;
    }

    switch (type) {
    case MirrorKind::Concave:  return std::make_unique<ConcaveMirror>(a, b, radius);
    case MirrorKind::Convex:   return std::make_unique<ConvexMirror>(a, b, radius);
    default: return std::make_unique<PlaneMirror>(a, b);
    }
}

// Одинаковое состояние луча: та же точка и то же направление
bool SameState(const Beam &a, const Beam &b)
{
    const double posEps = 1e-3;     // пиксели
    const double angEps = 1e-5;     // радианы

    if (std::abs(a.getX() - b.getX()) > posEps ||
        std::abs(a.getY() - b.getY()) > posEps)
        return false;

    return std::abs(std::remainder(a.getA() - b.getA(), Math::TwoPi)) <= angEps;
}

QString Fmt(double v)
{
    return QString::number(v, 'f', Style::MessageDecimals);
}

QString PointStr(const Points &p)
{
    return QString("(%1; %2)").arg(Fmt(p.getX()), Fmt(p.getY()));
}

QString TypeName(const Mirror &m)
{
    switch (TypeOf(m)) {
    case MirrorKind::Concave:  return QString("вогнутое, R = %1").arg(Fmt(m.getRadius()));
    case MirrorKind::Convex:   return QString("выпуклое, R = %1").arg(Fmt(m.getRadius()));
    default: return QString("плоское");
    }
}

// Точка зеркала по параметру t от 0 (начало) до 1 (конец)
Points PointAt(const Mirror &m, double t)
{
    const Points s = m.getStart();
    const Points e = m.getEnd();

    if (m.isFlat())
        return Points(s.getX() + t * (e.getX() - s.getX()),
                      s.getY() + t * (e.getY() - s.getY()));

    const Points c = m.CircleCenter();
    const double r = m.getRadius();

    const double a0 = std::atan2(s.getY() - c.getY(), s.getX() - c.getX());
    const double a1 = std::atan2(e.getY() - c.getY(), e.getX() - c.getX());

    // Меньшая дуга; центр слева от хорды - обход против часовой, справа - по часовой
    double delta = std::abs(std::remainder(a1 - a0, Math::TwoPi));
    if (m.getRad() < 0) delta = -delta;

    const double ang = a0 + t * delta;
    return Points(c.getX() + r * std::cos(ang), c.getY() + r * std::sin(ang));
}

// Наименьшее расстояние между двумя зеркалами: точки одного берутся
// с шагом, до другого считается точное расстояние (концы входят в выборку)
double WallDistance(const Mirror &a, const Mirror &b)
{
    double best = std::numeric_limits<double>::max();

    auto scan = [&](const Mirror &from, const Mirror &to) {
        for (int k = 0; k <= WallScanSteps; ++k) {
            double d = 0.0;
            to.ClosestPoint(PointAt(from, static_cast<double>(k) / WallScanSteps), d);
            best = std::min(best, d);
        }
    };

    scan(a, b);
    scan(b, a);
    return best;
}

// Соседние стены first -> second (конец first = начало second).
// Они не должны слипаться: дальние концы не ближе толщины линии к соседней стене.
bool AdjacentTooClose(const Mirror &first, const Mirror &second)
{
    double d = 0.0;
    second.ClosestPoint(first.getStart(), d);
    if (d < Style::WallWidth) return true;

    first.ClosestPoint(second.getEnd(), d);
    return d < Style::WallWidth;
}

enum class WallCheck { Ok, Cross, TooClose };

// Проверка набора стен: стены не должны пересекаться и касаться друг друга
// нигде, кроме общей вершины соседних стен (как Polygon::Check, но работает
// и для разомкнутой цепочки), а линии толщиной Style::WallWidth не должны
// сливаться. only - индексы изменённых стен: проверяются только пары с их
// участием (пусто - все пары). При ошибке в *a и *b - индексы стен (a < b).
WallCheck CheckWalls(const std::vector<const Mirror *> &w,
                     bool closed,
                     const std::vector<int> &only,
                     int *a,
                     int *b)
{
    const int n = static_cast<int>(w.size());

    auto involved = [&](int i) {
        if (only.empty()) return true;
        return std::find(only.begin(), only.end(), i) != only.end();
    };

    for (int i = 0; i < n; ++i) {
        for (int j = i + 1; j < n; ++j) {
            if (!involved(i) && !involved(j)) continue;

            const bool wrap = closed && i == 0 && j == n - 1;
            const bool adjacent = (j == i + 1) || wrap;

            // 1. Геометрические пересечения (кроме общей вершины соседей)
            std::vector<Points> allowed;
            if (j == i + 1) allowed.push_back(w[i]->getEnd());
            if (wrap)       allowed.push_back(w[j]->getEnd());

            for (const Points &p : w[i]->Intersections(*w[j])) {
                bool ok = false;
                for (const Points &q : allowed) {
                    if (p.Dist(q) < SamePointEps) { ok = true; break; }
                }
                if (!ok) {
                    if (a) *a = i;
                    if (b) *b = j;
                    return WallCheck::Cross;
                }
            }

            // 2. Толщина линий
            bool tooClose;
            if (adjacent) {
                const Mirror &first  = wrap ? *w[j] : *w[i];
                const Mirror &second = wrap ? *w[i] : *w[j];
                tooClose = AdjacentTooClose(first, second);
            } else {
                tooClose = WallDistance(*w[i], *w[j]) < Style::WallWidth;
            }

            if (tooClose) {
                if (a) *a = i;
                if (b) *b = j;
                return WallCheck::TooClose;
            }
        }
    }
    return WallCheck::Ok;
}

FrontEndOperators::FrontEndOperators() {}

void FrontEndOperators::Notify(NoteLevel level, const QString &text) const
{
    if (notify) notify(level, text);
}

void FrontEndOperators::CreatePoint(double x, double y) {
    // До замыкания стен на одну меньше: замыкающая добавится при замыкании комнаты
    if (static_cast<int>(prototype.sides_.size()) >= Limits::MaxWalls - 1) {
        Notify(NoteLevel::Warning,
               QString("Достигнут предел: в комнате не больше %1 сторон. "
                       "Новую точку поставить нельзя, замкните комнату.")
                   .arg(Limits::MaxWalls));
        return;
    }

    if (!prototype.isExist()) {
        prototype.isExist_ = true;
        prototype.first = Points(x, y);
        Notify(NoteLevel::Info,
               QString("Первая точка поставлена: %1.").arg(PointStr(Points(x, y))));
        return;
    }

    const Points from = prototype.sides_.empty()
                            ? prototype.first
                            : prototype.sides_.back()->getEnd();
    const Points to(x, y);

    try {
        // Новая стена не должна пересекать уже поставленные
        PlaneMirror wall(from, to);

        std::vector<const Mirror *> w;
        for (const auto &side : prototype.sides_)
            w.push_back(side.get());
        w.push_back(&wall);

        int a = 0, b = 0;
        const WallCheck res = CheckWalls(w, false, {static_cast<int>(w.size()) - 1}, &a, &b);
        if (res == WallCheck::Cross) {
            Notify(NoteLevel::Error,
                   QString("Нельзя поставить точку: новая стена пересекает или касается стены %1.")
                       .arg(a + 1));
            return;
        }
        if (res == WallCheck::TooClose) {
            Notify(NoteLevel::Error,
                   QString("Нельзя поставить точку: новая стена слишком близко к стене %1 "
                           "(ближе толщины линии, %2 пикселей).")
                       .arg(a + 1)
                       .arg(Style::WallWidth));
            return;
        }

        prototype.sides_.push_back(std::make_unique<PlaneMirror>(from, to));
    } catch (const std::exception &e) {
        Notify(NoteLevel::Error,
               QString("Не удалось поставить точку: %1").arg(QString::fromUtf8(e.what())));
        return;
    }

    Notify(NoteLevel::Info,
           QString("Точка %1 поставлена: %2.")
               .arg(static_cast<int>(prototype.sides_.size()) + 1)
               .arg(PointStr(to)));
}

void FrontEndOperators::Binding() {
    ConnectPoints(0);
}

void FrontEndOperators::ConnectPoints(int num)
{
    if (num < 0 || num >= static_cast<int>(prototype.sides_.size()))
        return;

    // Комната должна остаться многоугольником: не меньше Limits::MinWalls сторон
    // (первые num стен при соединении удаляются, добавляется замыкающая)
    const int resultWalls = static_cast<int>(prototype.sides_.size()) - num + 1;
    if (resultWalls < Limits::MinWalls) {
        Notify(NoteLevel::Error,
               QString("Нельзя замкнуть комнату: получится сторон: %1, а нужно от %2 до %3. "
                       "Подключите более раннюю точку или поставьте ещё точки.")
                   .arg(resultWalls)
                   .arg(Limits::MinWalls)
                   .arg(Limits::MaxWalls));
        return;
    }

    Points start = prototype.sides_[num]->getStart();
    Points end = prototype.sides_.back()->getEnd();

    // Замыкающая стена нулевой длины невозможна
    if (start.Dist(end) < SamePointEps) {
        Notify(NoteLevel::Error,
               "Нельзя замкнуть комнату: начальная и конечная точки совпадают.");
        return;
    }

    // Замыкающая стена не должна пересекать остальные
    {
        PlaneMirror closing(end, start);

        std::vector<const Mirror *> w;
        for (int i = num; i < static_cast<int>(prototype.sides_.size()); ++i)
            w.push_back(prototype.sides_[i].get());
        w.push_back(&closing);

        int a = 0, b = 0;
        const WallCheck res = CheckWalls(w, true, {static_cast<int>(w.size()) - 1}, &a, &b);
        if (res == WallCheck::Cross) {
            Notify(NoteLevel::Error,
                   QString("Нельзя замкнуть комнату: замыкающая стена пересекает или касается стены %1.")
                       .arg(a + num + 1));
            return;
        }
        if (res == WallCheck::TooClose) {
            Notify(NoteLevel::Error,
                   QString("Нельзя замкнуть комнату: замыкающая стена слишком близко к стене %1 "
                           "(ближе толщины линии, %2 пикселей).")
                       .arg(a + num + 1)
                       .arg(Style::WallWidth));
            return;
        }
    }

    // Луч: у стёртых стен он удаляется, у остальных сдвигаются индексы
    if (prototype.isBeamSet_) {
        if (prototype.beamWall_ < num) {
            prototype.isBeamSet_ = false;
            Notify(NoteLevel::Warning,
                   "Источник луча удалён: стена, на которой он стоял, исчезла при соединении точек.");
        } else {
            prototype.beamWall_ -= num;
        }
    }

    prototype.sides_.erase(
        prototype.sides_.begin(),
        prototype.sides_.begin() + num
        );
    prototype.first = start;

    prototype.sides_.push_back(std::make_unique<PlaneMirror>(end, start));
    prototype.isCycled_ = true;

    Notify(NoteLevel::Info,
           QString("Комната замкнута, сторон: %1.")
               .arg(static_cast<int>(prototype.sides_.size())));
}

void FrontEndOperators::DeleteLast() {
    if (prototype.sides_.empty()) {
        if (prototype.isExist()) {
            prototype.isExist_ = false;
            Notify(NoteLevel::Info,
                   QString("Первая точка 1 %1 удалена.").arg(PointStr(prototype.first)));
        } else {
            Notify(NoteLevel::Warning, "Откатывать нечего: точек нет.");
        }
    } else {
        const int last = static_cast<int>(prototype.sides_.size()) - 1;
        const bool wasCycled = prototype.isCycled_;

        if (prototype.isBeamSet_ && prototype.beamWall_ == last) {
            prototype.isBeamSet_ = false;
            Notify(NoteLevel::Warning,
                   "Источник луча удалён вместе со стеной, на которой он стоял.");
        }

        const Points lastPoint = prototype.sides_.back()->getEnd();
        const int lastPointNo = static_cast<int>(prototype.sides_.size()) + 1;

        prototype.sides_.pop_back();
        prototype.isCycled_ = false;

        Notify(NoteLevel::Info,
               wasCycled ? QString("Замыкающая стена удалена: комната снова не замкнута.")
                         : QString("Последняя точка %1 %2 удалена.")
                               .arg(lastPointNo)
                               .arg(PointStr(lastPoint)));
    }
}

void FrontEndOperators::DeletePoint(int num)
{
    auto &sides = prototype.sides_;
    const int n = static_cast<int>(sides.size());
    const bool cycled = prototype.isCycled_;

    // Замкнутая комната: вершин ровно столько же, сколько стен (num: 0..n-1)
    if (!prototype.isExist() || num < 0 || num > n || (cycled && num >= n)) {
        Notify(NoteLevel::Error, "Не удалось удалить точку: такой точки нет.");
        return;
    }

    // После удаления в замкнутой комнате должно остаться не меньше Limits::MinWalls вершин
    if (cycled && n - 1 < Limits::MinWalls) {
        Notify(NoteLevel::Error,
               QString("Нельзя удалить точку: в замкнутой комнате должно остаться не меньше %1 вершин "
                       "(сейчас их %2).")
                   .arg(Limits::MinWalls)
                   .arg(n));
        return;
    }

    const int pointNo = num + 1;
    const Points oldPoint = (num < n)  ? sides[num]->getStart()
                            : (n > 0)  ? sides.back()->getEnd()
                                      : prototype.first;

    // Если на исчезающей стене (с нуля) стоял луч - он удаляется
    auto dropBeamIfOn = [&](int lo, int hi) {
        if (prototype.isBeamSet_ && prototype.beamWall_ >= lo && prototype.beamWall_ <= hi) {
            prototype.isBeamSet_ = false;
            Notify(NoteLevel::Warning,
                   "Источник луча удалён вместе со стеной, на которой он стоял.");
        }
    };

    if (!cycled) {
        // 1. Единственная точка: комнаты больше нет
        if (n == 0) {
            prototype.isExist_ = false;
            prototype.first = Points();
            Notify(NoteLevel::Info,
                   QString("Единственная точка 1 %1 удалена.").arg(PointStr(oldPoint)));
            return;
        }

        // 2. Последняя точка: исчезает последняя стена
        if (num == n) {
            dropBeamIfOn(n - 1, n - 1);
            sides.pop_back();
            Notify(NoteLevel::Info,
                   QString("Точка %1 %2 удалена вместе со стеной %3.")
                       .arg(pointNo)
                       .arg(PointStr(oldPoint))
                       .arg(n));
            return;
        }

        // 3. Первая точка: исчезает первая стена, первой становится её конец
        if (num == 0) {
            const Points newFirst = sides[0]->getEnd();
            dropBeamIfOn(0, 0);
            if (prototype.isBeamSet_)
                --prototype.beamWall_;   // все стены сдвигаются на одну
            sides.erase(sides.begin());
            prototype.first = newFirst;
            Notify(NoteLevel::Info,
                   QString("Точка 1 %1 удалена вместе со стеной 1. Первой стала бывшая точка 2 %2.")
                       .arg(PointStr(oldPoint))
                       .arg(PointStr(newFirst)));
            return;
        }
    }

    // 4. Вершина с двумя соседями (средняя точка незамкнутой комнаты или любая
    //    вершина замкнутой): стены prev и num заменяются одной плоской стеной между соседями
    const int prev = (num - 1 + n) % n;
    const Points a = sides[prev]->getStart();
    const Points b = sides[num]->getEnd();

    std::unique_ptr<PlaneMirror> merged;
    try {
        merged = std::make_unique<PlaneMirror>(a, b);
    } catch (const std::exception &e) {
        Notify(NoteLevel::Error,
               QString("Не удалось удалить точку %1: %2")
                   .arg(pointNo)
                   .arg(QString::fromUtf8(e.what())));
        return;
    }

    // Набор стен после удаления: oldIdx[k] - прежний индекс стены (-1 - новая стена)
    std::vector<const Mirror *> w;
    std::vector<int> oldIdx;
    int mergedPos = 0;

    if (cycled && num == 0) {
        // стена 0 исчезает, на месте последней стоит новая (она заканчивается там же, где кончалась стена 0)
        for (int i = 1; i < n - 1; ++i) {
            w.push_back(sides[i].get());
            oldIdx.push_back(i);
        }
        mergedPos = static_cast<int>(w.size());
        w.push_back(merged.get());
        oldIdx.push_back(-1);
    } else {
        for (int i = 0; i < n; ++i) {
            if (i == num) continue;
            if (i == num - 1) {
                mergedPos = static_cast<int>(w.size());
                w.push_back(merged.get());
                oldIdx.push_back(-1);
            } else {
                w.push_back(sides[i].get());
                oldIdx.push_back(i);
            }
        }
    }

    // Номера соседних точек для сообщений (с единицы)
    const int prevPointNo = prev + 1;
    const int nextPointNo = cycled ? (num + 1) % n + 1 : num + 2;

    int x = 0, y = 0;
    const WallCheck res = CheckWalls(w, cycled, {mergedPos}, &x, &y);
    if (res != WallCheck::Ok) {
        const int other = (x == mergedPos) ? y : x;
        const int otherNo = oldIdx[other] + 1;   // номер в прежней нумерации

        if (res == WallCheck::Cross) {
            Notify(NoteLevel::Error,
                   QString("Нельзя удалить точку %1: новая стена между точками %2 и %3 "
                           "пересечёт или коснётся стены %4.")
                       .arg(pointNo)
                       .arg(prevPointNo)
                       .arg(nextPointNo)
                       .arg(otherNo));
        } else {
            Notify(NoteLevel::Error,
                   QString("Нельзя удалить точку %1: новая стена между точками %2 и %3 "
                           "окажется слишком близко к стене %4 (ближе толщины линии, %5 пикселей).")
                       .arg(pointNo)
                       .arg(prevPointNo)
                       .arg(nextPointNo)
                       .arg(otherNo)
                       .arg(Style::WallWidth));
        }
        return;
    }

    // Применяем
    dropBeamIfOn(prev, prev);
    dropBeamIfOn(num, num);

    if (prototype.isBeamSet_) {
        // индекс стены с лучом в новой нумерации
        for (int j = 0; j < static_cast<int>(oldIdx.size()); ++j) {
            if (oldIdx[j] == prototype.beamWall_) {
                prototype.beamWall_ = j;
                break;
            }
        }
    }

    std::vector<std::unique_ptr<Mirror>> result;
    result.reserve(oldIdx.size());
    for (int k : oldIdx) {
        if (k < 0) result.push_back(std::move(merged));
        else       result.push_back(std::move(sides[k]));
    }
    sides = std::move(result);

    if (cycled && num == 0)
        prototype.first = sides[0]->getStart();   // первой стала прежняя вторая вершина

    Notify(NoteLevel::Info,
           QString("Точка %1 %2 удалена. Стены %3 и %4 заменены одной плоской стеной между "
                   "бывшими точками %5 %6 и %7 %8.")
               .arg(pointNo)
               .arg(PointStr(oldPoint))
               .arg(prev + 1)
               .arg(num + 1)
               .arg(prevPointNo)
               .arg(PointStr(a))
               .arg(nextPointNo)
               .arg(PointStr(b)));
}

void FrontEndOperators::ChangeMirror(
    int num,
    const ChangePointData &data)
{
    auto &sides = prototype.sides_;
    const int n = static_cast<int>(sides.size());
    const bool cycled = prototype.isCycled_;

    if (n == 0 || num < 0 || num > n || (cycled && num == n)) {
        Notify(NoteLevel::Error, "Не удалось изменить точку: такой точки нет.");
        return;
    }

    // Последняя точка незамкнутого полигона - конец последней стены
    const bool lastPoint = (!cycled && num == n);
    const Points oldPoint = lastPoint ? sides.back()->getEnd()
                                      : sides[num]->getStart();
    const int pointNo = num + 1;

    // ---------------------------------------------------------
    // 1. Новые координаты: вне допустимой зоны - оставляем прежние
    // ---------------------------------------------------------
    double newX = data.x;
    double newY = data.y;

    if (!(newX >= Limits::MinX && newX <= Limits::MaxX)) {
        Notify(NoteLevel::Error,
               QString("Координата X = %1 вне допустимой зоны (от %2 до %3): оставлена прежней (%4).")
                   .arg(Fmt(data.x), Fmt(Limits::MinX), Fmt(Limits::MaxX), Fmt(oldPoint.getX())));
        newX = oldPoint.getX();
    }
    if (!(newY >= Limits::MinY && newY <= Limits::MaxY)) {
        Notify(NoteLevel::Error,
               QString("Координата Y = %1 вне допустимой зоны (от %2 до %3): оставлена прежней (%4).")
                   .arg(Fmt(data.y), Fmt(Limits::MinY), Fmt(Limits::MaxY), Fmt(oldPoint.getY())));
        newY = oldPoint.getY();
    }

    Points newPoint(newX, newY);

    // ---------------------------------------------------------
    // 2. Слишком близко к другой вершине - координаты не меняем
    // ---------------------------------------------------------
    const int vertexCount = cycled ? n : n + 1;
    for (int j = 0; j < vertexCount; ++j) {
        if (j == num) continue;

        const Points v = (j < n) ? sides[j]->getStart()
                                 : sides.back()->getEnd();

        if (v.Dist(newPoint) <= Limits::MinPointDist) {
            Notify(NoteLevel::Error,
                   QString("Точка %1 слишком близко к точке %2 (ближе %3 пикселей): координаты не изменены.")
                       .arg(pointNo)
                       .arg(j + 1)
                       .arg(static_cast<int>(Limits::MinPointDist)));
            newPoint = oldPoint;
            break;
        }
    }

    // ---------------------------------------------------------
    // 3. Собираем новые стены, пока ничего не меняя в полигоне
    // ---------------------------------------------------------
    struct Item {
        int idx;
        std::unique_ptr<Mirror> mirror;
        bool flattened;
        double halfChord;
        double radius;
    };
    std::vector<Item> items;

    // Стена сохраняет свой тип и радиус, меняются только концы
    auto addKeep = [&](int idx, const Points &a, const Points &b) {
        Item it{idx, nullptr, false, a.Dist(b) / 2.0, sides[idx]->getRadius()};
        it.mirror = MakeMirror(TypeOf(*sides[idx]), a, b,
                               sides[idx]->getRadius(), &it.flattened);
        items.push_back(std::move(it));
    };

    // Стена получает тип и радиус из диалога
    auto addNew = [&](int idx, const Points &a, const Points &b) {
        Item it{idx, nullptr, false, a.Dist(b) / 2.0, data.radius};
        it.mirror = MakeMirror(data.mirrorType, a, b, data.radius, &it.flattened);
        items.push_back(std::move(it));
    };

    try {
        if (lastPoint) {
            addKeep(n - 1, sides[n - 1]->getStart(), newPoint);
        } else if (cycled && num == 0) {
            addNew(0, newPoint, sides[0]->getEnd());
            addKeep(n - 1, sides[n - 1]->getStart(), newPoint);
        } else {
            addNew(num, newPoint, sides[num]->getEnd());
            if (num > 0)
                addKeep(num - 1, sides[num - 1]->getStart(), newPoint);
        }
    } catch (const std::exception &e) {
        Notify(NoteLevel::Error,
               QString("Не удалось изменить точку %1: %2")
                   .arg(pointNo)
                   .arg(QString::fromUtf8(e.what())));
        return;
    }

    // ---------------------------------------------------------
    // 3б. Новые стены не должны пересекаться с остальными
    // ---------------------------------------------------------
    {
        std::vector<const Mirror *> w(n);
        std::vector<int> changed;

        for (int i = 0; i < n; ++i)
            w[i] = sides[i].get();
        for (const Item &it : items) {
            w[it.idx] = it.mirror.get();
            changed.push_back(it.idx);
        }

        int a = 0, b = 0;
        const WallCheck res = CheckWalls(w, cycled, changed, &a, &b);
        if (res == WallCheck::Cross) {
            Notify(NoteLevel::Error,
                   QString("Изменение точки %1 отменено: зеркала %2 и %3 начнут пересекаться или касаться.")
                       .arg(pointNo)
                       .arg(a + 1)
                       .arg(b + 1));
            return;
        }
        if (res == WallCheck::TooClose) {
            Notify(NoteLevel::Error,
                   QString("Изменение точки %1 отменено: зеркала %2 и %3 окажутся слишком близко "
                           "(ближе толщины линии, %4 пикселей).")
                       .arg(pointNo)
                       .arg(a + 1)
                       .arg(b + 1)
                       .arg(Style::WallWidth));
            return;
        }
    }

    // ---------------------------------------------------------
    // 4. Луч стоял на изменяемой стене - он удаляется
    // ---------------------------------------------------------
    if (prototype.isBeamSet_) {
        for (const Item &it : items) {
            if (it.idx == prototype.beamWall_) {
                prototype.isBeamSet_ = false;
                Notify(NoteLevel::Warning,
                       "Источник луча удалён: стена, на которой он стоял, изменена.");
                break;
            }
        }
    }

    // ---------------------------------------------------------
    // 5. Применяем изменения
    // ---------------------------------------------------------
    for (Item &it : items)
        sides[it.idx] = std::move(it.mirror);

    if (num == 0)
        prototype.first = newPoint;

    // ---------------------------------------------------------
    // 6. Сообщения
    // ---------------------------------------------------------
    for (const Item &it : items) {
        if (it.flattened) {
            Notify(NoteLevel::Warning,
                   QString("Зеркало %1: радиус %2 меньше половины его длины (%3), зеркало оставлено плоским.")
                       .arg(it.idx + 1)
                       .arg(Fmt(it.radius), Fmt(it.halfChord)));
        }
    }

    if (newPoint.Dist(oldPoint) > NoMoveEps) {
        Notify(NoteLevel::Info,
               QString("Точка %1 перемещена: %2 -> %3.")
                   .arg(pointNo)
                   .arg(PointStr(oldPoint), PointStr(newPoint)));
    }

    if (!lastPoint) {
        Notify(NoteLevel::Info,
               QString("Зеркало %1: %2.")
                   .arg(items[0].idx + 1)
                   .arg(TypeName(*sides[items[0].idx])));
    }
}

// Запоминает комнату (вершины, типы и радиусы стен) и параметры источника луча
TraceSnapshot FrontEndOperators::MakeSnapshot() const
{
    TraceSnapshot s;

    for (const auto &m : prototype.sides_) {
        TraceSnapshot::Wall w;
        w.x = m->getStart().getX();
        w.y = m->getStart().getY();
        w.type = TypeOf(*m);
        w.radius = std::abs(m->getRadius());
        s.walls.push_back(w);
    }

    s.beamWall = static_cast<int>(prototype.beamWall_);
    s.beamWallAngle = prototype.beamWallAngle_;
    return s;
}

bool FrontEndOperators::NextReflection()
{
    if (!prototype.isBeamSet_) {
        Notify(NoteLevel::Error, "Сначала поставьте источник луча.");
        return false;
    }

    if (looped_) {
        Notify(NoteLevel::Warning,
               "Луч уже зациклился: дальше отражения только повторяются.");
        return false;
    }

    // Первое отражение (или первое после сброса): только теперь стираем прежнюю
    // историю и идём от стартового луча полигона
    if (traceCleared_ || history_.size() <= 1) {
        tempBeam_ = prototype.beam_;
        history_.assign(1, Points(tempBeam_.getX(), tempBeam_.getY()));
        states_.assign(1, tempBeam_);
        snapshot_ = MakeSnapshot();
        traceCleared_ = false;
    }

    auto result = prototype.TraceBeamIter(tempBeam_);
    if (result.second.Dist(history_.back()) < NoMoveEps) {
        Notify(NoteLevel::Error,
               "Не удалось найти следующее отражение: луч никуда не попал.");
        return false;
    }

    tempBeam_ = result.first;
    history_.push_back(result.second);
    states_.push_back(tempBeam_);

    const int step = static_cast<int>(history_.size()) - 1;

    Notify(NoteLevel::Reflection,
           QString("Отражение %1: %2.").arg(step).arg(PointStr(result.second)));

    // Такая точка с таким направлением уже была - ход луча повторяется
    for (int m = 0; m < step; ++m) {
        if (SameState(states_[m], tempBeam_)) {
            looped_ = true;
            Notify(NoteLevel::Warning,
                   QString("Луч света зациклился: после отражения %1 ход луча повторяется "
                           "(период - %2). Дальше отражаться нечего: измените угол луча "
                           "(клик по красной точке) или переставьте источник.")
                       .arg(step)
                       .arg(step - m));
            break;
        }
    }

    return true;
}

void FrontEndOperators::ResetTrace()
{
    // Путь исчезает с экрана сразу (traceCleared_), но history_ и states_ остаются
    // нетронутыми до первого нового отражения (см. NextReflection)
    traceCleared_ = true;
    looped_ = false;
}

struct TraceSnapshot
{
    struct Wall
    {
        double x = 0.0;        // начало стены (вершина)
        double y = 0.0;
        int    type = MirrorKind::Plane;   // см. MirrorKind
        double radius = 0.0;   // радиус кривизны (для плоской не важен)
    };

    std::vector<Wall> walls;
    int    beamWall = -1;          // индекс стены с источником (с нуля)

};

class FrontEndOperators
{
public:
    FrontEndOperators();

    void CreatePoint(double x, double y);
    void ConnectPoints(int num);
    void ChangeMirror(int num, const ChangePointData &data);
    void DeleteLast();
    // Удаляет точку незамкнутой комнаты (num: 0 - первая, n - последняя, как в ChangeMirror):
    //  - единственная точка просто исчезает;
    //  - крайняя точка исчезает вместе со своей стеной;
    //  - у средней точки обе стены заменяются одной плоской стеной между её соседями
    //    (если она пересекает другие стены или слишком близко к ним - удаление отменяется)
    //  - в замкнутой комнате можно удалить любую вершину (стены по обе стороны от неё
    //    заменяются одной плоской), но после удаления должно остаться не меньше Limits::MinWalls
    void DeletePoint(int num);
    void Binding();

    // Пошаговая трассировка луча
    bool NextReflection();   // следующая точка отражения (false - не удалось)
    void ResetTrace();       // сбросить путь: на экране его нет сразу, а history_ очищается
        // только при первом новом отражении (traceCleared_)
    bool HasTrace() const { return !traceCleared_ && history_.size() > 1; }   // есть ли путь для показа

    // Сюда FrontEndOperators шлёт сообщения для служебного окна
    std::function<void(NoteLevel, const QString &)> notify;

    friend class MainWindow;

private:
    void Notify(NoteLevel level, const QString &text) const;
    TraceSnapshot MakeSnapshot() const;   // комната и луч на данный момент

    Polygon prototype;
    Beam tempBeam_;
    std::vector<Points> history_;   // стартовая точка + точки отражений
    std::vector<Beam> states_;      // состояния луча (точка + угол): старт + после каждого отражения
    bool looped_ = false;           // ход луча повторился - луч зациклился
    bool traceCleared_ = false;     // путь сброшен: не показывается, но history_ хранит старые данные

    TraceSnapshot snapshot_;        // комната и луч на момент старта той симуляции, чья история в history_
};

// Доступность кнопок (недоступные приглушаются):
//  - в режиме выбора точки луча недоступны кнопки 2, 3, 4;
//  - если луч зациклился, недоступны кнопки 3 и 4;
//  - во время анимации и повтора кнопки приглушены, но остаются нажимаемыми:
//    нажатие игнорируется с сообщением в чат;
//  - во время повтора кнопка 4 работает как "Остановить повтор", а после
//    запроса остановки приглушается тоже, пока не закончится анимация.
void MainWindow::UpdateButtons()
{
    const bool picking = !Redacting && Beaming;
    const bool looped = polygonCreator.looped_;
    const bool busy = animating_ || repeating_;

    // enabled - принимает ли кнопка нажатия, dim - приглушена ли она
    auto setState = [](QPushButton *b, bool enabled, bool dim)
    {
        b->setEnabled(enabled);

        if (dim)
        {
            auto *effect = new QGraphicsOpacityEffect(b);
            effect->setOpacity(Style::DimOpacity);
            b->setGraphicsEffect(effect);   // старый эффект Qt удаляет сам
        }
        else
        {
            b->setGraphicsEffect(nullptr);
        }
    };

    setState(ui->pushButton,   true,                busy);
    setState(ui->pushButton_2, !picking,            picking || busy);
    setState(ui->pushButton_3, !picking && !looped, picking || looped || busy);

    if (repeating_)
    {
        setState(ui->pushButton_4, true, stopRequested_);
        ui->pushButton_4->setText(stopRequested_ ? "Останавливаю..." : "Остановить повтор");
    }
    else
    {
        setState(ui->pushButton_4, !picking && !looped, picking || looped || animating_);
        ui->pushButton_4->setText("Повтор анимации");
    }

    // Кнопка 5 (сохранение истории): недоступна, пока истории нет или выбирается источник;
    // во время анимации/повтора приглушена, нажатие игнорируется с сообщением в чат.
    // Текст: "текущая" история, пока путь на экране, и "прошлой симуляции" после сброса.
    const bool hasHistory = polygonCreator.history_.size() > 1;
    setState(ui->pushButton_5, hasHistory && !picking, !hasHistory || picking || busy);
    ui->pushButton_5->setText(polygonCreator.HasTrace()
                                  ? "Сохранить текущую историю отражений"
                                  : "Сохранить историю прошлой симуляции");

    // Ползунок скорости и его подпись: тускнеют, когда менять скорость нельзя.
    // Сам ползунок остаётся enabled, чтобы eventFilter мог показать сообщение в чат.
    auto dimWidget = [](QWidget *w, bool on)
    {
        if (on)
        {
            auto *effect = new QGraphicsOpacityEffect(w);
            effect->setOpacity(Style::DimOpacity);
            w->setGraphicsEffect(effect);   // старый эффект Qt удаляет сам
        }
        else
        {
            w->setGraphicsEffect(nullptr);
        }
    };
    dimWidget(ui->horizontalSlider, SettingsBlocked());
    dimWidget(ui->label, SettingsBlocked());
    dimWidget(ui->horizontalSlider_2, SettingsBlocked());
    dimWidget(ui->label_2, SettingsBlocked());
    dimWidget(ui->checkBox, SettingsBlocked());
    dimWidget(ui->label_3, TraceLimitBlocked());
    dimWidget(ui->textEdit, TraceLimitBlocked());
}

// Служебное окно: добавляет сообщение-"пузырь" сверху, старые сдвигаются вниз
void MainWindow::AddMessage(const QString &text, MsgType type)
{
    QString bg;
    QString tag;

    switch (type)
    {
    case MsgType::Error:
        bg = "#5c2b2e";
        tag = "Ошибка";
        break;
    case MsgType::Warning:
        bg = "#5c4b1f";
        tag = "Внимание";
        break;
    case MsgType::Reflection:
        bg = "#24512f";
        tag = "Отражение";
        break;
    default:
        bg = "#303134";
        tag = "Система";
        break;
    }

    QString body = text.toHtmlEscaped();
    body.replace("\n", "<br>");

    auto *bubble = new QLabel(
        QString("<span style='color:#9aa0a6; font-size:12px;'>%1 · %2</span><br>%3")
            .arg(tag, QTime::currentTime().toString("HH:mm:ss"), body)
        );

    bubble->setTextFormat(Qt::RichText);
    bubble->setWordWrap(true);
    bubble->setTextInteractionFlags(Qt::TextSelectableByMouse);
    bubble->setStyleSheet(QString(
                              "QLabel {"
                              " background-color: %1;"
                              " color: white;"
                              " border-radius: 12px;"
                              " padding: 8px 12px;"
                              " font-size: 14px;"
                              "}").arg(bg));

    // новое сообщение - в самый верх, старые уходят вниз
    // (растяжка в конце прижимает все сообщения к верху)
    chatLayout->insertWidget(0, bubble);

    // Окошко не разрастается бесконечно: самые старые сообщения (внизу) удаляем
    while (chatLayout->count() - 1 > Style::MaxChatMessages)
    {
        QLayoutItem *old = chatLayout->takeAt(chatLayout->count() - 2);
        delete old->widget();
        delete old;
    }
}

}

void MainWindow::SecondButClicked() {
    if (IgnoreDuringAnimation())
        return;

    if (Redacting) {
        auto &poly = polygonCreator.prototype;

        // Сторон в замкнутой комнате: у незамкнутой добавится замыкающая
        const int finalWalls = static_cast<int>(poly.sides_.size()) + (poly.isCycled_ ? 0 : 1);
        if (finalWalls < Limits::MinWalls) {
            AddMessage(
                QString("Нельзя начать симуляцию: слишком мало сторон. В комнате должно быть "
                        "от %1 до %2 сторон, а сейчас получится %3. Поставьте ещё точки.")
                    .arg(Limits::MinWalls)
                    .arg(Limits::MaxWalls)
                    .arg(finalWalls),
                MsgType::Error);
            return;
        }

        // Уже замкнутый полигон повторно не замыкаем
        if (!poly.isCycled_)
            polygonCreator.Binding();
        update();

        // Замкнуть не удалось (причина уже в чате)
        if (!poly.isCycled_)
            return;

        if (!poly.Check()) {
            AddMessage(
                "Комната некорректна: стены пересекаются или касаются "
                "друг друга. Исправьте точки или откатите последнюю.",
                MsgType::Error);
            return;
        }

        Redacting = false;
        ui->pushButton->setText("Редактировать комнату");
        ui->pushButton_2->setText("Переставить источник света");
        ui->pushButton_3->setVisible(true);
        ui->pushButton_4->setVisible(true);
        ShowSimControls(true);

        // форма комнаты могла измениться - пересчитываем направление луча
        if (poly.isBeamSet_)
            poly.SetBeamAngle(poly.beamWallAngle_);
        polygonCreator.ResetTrace();

        // если луч пережил редактирование, выбирать заново не нужно
        Beaming = !poly.isBeamSet_;
        UpdateButtons();

        // Вершины комнаты и типы стен
        QString desc = QString("Симуляция начата. Комната замкнута, сторон: %1.")
                           .arg(static_cast<int>(poly.sides_.size()));
        for (size_t i = 0; i < poly.sides_.size(); ++i)
        {
            const Mirror &m = *poly.sides_[i];

            QString kind = "плоская";
            if (dynamic_cast<const ConcaveMirror *>(&m))
                kind = QString("вогнутая, R = %1").arg(m.getRadius(), 0, 'f', Style::MessageDecimals);
            else if (dynamic_cast<const ConvexMirror *>(&m))
                kind = QString("выпуклая, R = %1").arg(m.getRadius(), 0, 'f', Style::MessageDecimals);

            desc += QString("\nТочка %1: (%2; %3), стена %1 - %4")
                        .arg(static_cast<int>(i) + 1)
                        .arg(m.getStart().getX(), 0, 'f', Style::MessageDecimals)
                        .arg(m.getStart().getY(), 0, 'f', Style::MessageDecimals)
                        .arg(kind);
        }
        AddMessage(desc);

        if (Beaming)
        {
            AddMessage("Кликните по стене, откуда должен выйти луч.");
        }
        else
        {
            AddMessage(QString("Источник луча сохранён: (%1; %2) на стене %3, угол к стене %4°.")
                           .arg(poly.beam_.getX(), 0, 'f', Style::MessageDecimals)
                           .arg(poly.beam_.getY(), 0, 'f', Style::MessageDecimals)
                           .arg(poly.beamWall_ + 1)
                           .arg(poly.beamWallAngle_, 0, 'f', BeamAngle::Decimals));
        }
    } else {
        // Новый источник: старый луч удаляем сразу, ещё до выбора нового
        polygonCreator.prototype.isBeamSet_ = false;
        polygonCreator.ResetTrace();
        Beaming = true;
        UpdateButtons();
        update();
        AddMessage("Источник луча удалён. Кликните по стене, откуда должен выйти новый луч.");
    }
}

void MainWindow::ThirdButClicked() {
    if (IgnoreDuringAnimation())
        return;

    if (Redacting || Beaming || !polygonCreator.prototype.isBeamSet_)
        return;

    StartReflectionStep();
}

// Кнопка 4: повтор анимации. Первое нажатие запускает отражения одно за другим,
// второе просит остановиться после текущей анимации, а до её конца кнопка
// (как и все остальные) нажатия игнорирует.
void MainWindow::FourthButClicked()
{
    if (repeating_)
    {
        // Идёт пауза между отражениями: останавливаем сразу, паузу сбрасываем,
        // следующее отражение не делаем, все кнопки сразу снова активны
        if (delaying_)
        {
            delayTimer_->stop();
            delaying_ = false;
            repeating_ = false;
            stopRequested_ = false;
            AddMessage("Повтор остановлен: пауза между отражениями прервана.");
            UpdateButtons();
            update();
            return;
        }

        if (stopRequested_)
        {
            AddMessage(
                "Остановка повтора уже запрошена: дождитесь конца анимации.",
                MsgType::Warning);
            return;
        }

        stopRequested_ = true;
        AddMessage("Повтор остановится после завершения текущей анимации.");
        UpdateButtons();
        return;
    }

    // Одиночная анимация (кнопка 3) ещё идёт
    if (IgnoreDuringAnimation())
        return;

    if (Redacting || Beaming || !polygonCreator.prototype.isBeamSet_
        || polygonCreator.looped_)
        return;

    repeating_ = true;
    stopRequested_ = false;

    if (!StartReflectionStep())
    {
        // первый шаг не получился (причина уже в чате)
        repeating_ = false;
        UpdateButtons();
        return;
    }

    AddMessage(
        "Повтор запущен: луч отражается сам. Нажмите кнопку ещё раз, "
        "чтобы остановить (или повтор закончится, когда луч зациклится).");
}

// Кнопка 5: сохраняет всю историю отражений (history_) в txt-файл, выбранный пользователем.
// История хранится и после сброса пути, поэтому можно сохранить и прошлую симуляцию.
// В файл попадают комната и параметры луча на момент старта той симуляции (snapshot_).
void MainWindow::FifthButClicked()
{
    if (animating_ || repeating_)
    {
        AddMessage(
            "Нельзя сохранить историю, пока идёт анимация или повтор. "
            "Дождитесь конца анимации или остановите повтор.",
            MsgType::Warning);
        return;
    }

    if (!Redacting && Beaming)
    {
        AddMessage(
            "Нельзя сохранить историю, пока не выбран источник на стене.",
            MsgType::Warning);
        return;
    }

    // копии на время диалогов
    const std::vector<Points> hist = polygonCreator.history_;
    const std::vector<Beam> states = polygonCreator.states_;
    const TraceSnapshot snap = polygonCreator.snapshot_;

    if (hist.size() <= 1)
    {
        AddMessage("Сохранять нечего: история отражений пуста.", MsgType::Warning);
        return;
    }

    const bool current = polygonCreator.HasTrace();

    // 1) Сколько знаков после запятой сохранять в координатах
    SaveHistoryDialog precisionDialog(
        saveDecimals_, Style::SaveDecimalsMax,
        hist[1].getX(), hist[1].getY(), this);
    precisionDialog.move(
        this->geometry().center()
        - precisionDialog.rect().center()
        );

    if (precisionDialog.exec() != QDialog::Accepted)
    {
        AddMessage("Сохранение истории отменено.");
        return;
    }
    saveDecimals_ = precisionDialog.getDecimals();

    // 2) Куда сохранять
    if (lastSaveDir_.isEmpty())
        lastSaveDir_ = QStandardPaths::writableLocation(QStandardPaths::DocumentsLocation);

    const QString defaultName =
        lastSaveDir_ + "/reflections_"
        + QDateTime::currentDateTime().toString("yyyyMMdd_HHmmss") + ".txt";

    QString path = QFileDialog::getSaveFileName(
        this, "Сохранить историю отражений", defaultName, "Текстовые файлы (*.txt)");

    if (path.isEmpty())
    {
        AddMessage("Сохранение истории отменено.");
        return;
    }

    // Файл всегда .txt; если пришлось дописать расширение, перезапись не подтверждалась
    // диалогом, поэтому существующий файл не трогаем
    if (QFileInfo(path).suffix().compare("txt", Qt::CaseInsensitive) != 0)
    {
        path += ".txt";
        if (QFileInfo::exists(path))
        {
            AddMessage(
                QString("Файл «%1» уже существует. Выберите другое имя или укажите "
                        "расширение .txt, чтобы перезаписать файл.").arg(path),
                MsgType::Error);
            return;
        }
    }

    lastSaveDir_ = QFileInfo(path).absolutePath();

    QFile file(path);
    if (!file.open(QIODevice::WriteOnly | QIODevice::Text | QIODevice::Truncate))
    {
        AddMessage(
            QString("Не удалось открыть файл для записи: %1 (%2).")
                .arg(path, file.errorString()),
            MsgType::Error);
        return;
    }

    // Форматирование чисел
    const int dec = saveDecimals_;
    auto num = [dec](double v) { return QString::number(v, 'f', dec); };
    auto pt = [&num](double x, double y)
    {
        return QString("(%1; %2)").arg(num(x), num(y));
    };
    // Угол луча (радианы) -> градусы в пределах [0; 360)
    auto ang = [](double rad)
    {
        double d = std::fmod(rad * Math::HalfTurnDeg / Math::Pi, Math::FullTurnDeg);
        if (d < 0.0)
            d += Math::FullTurnDeg;
        return QString::number(d, 'f', Style::SaveAngleDecimals);
    };

    QTextStream out(&file);
    out.setEncoding(QStringConverter::Utf8);

    out << "История отражений луча\n";
    out << "Сохранено: "
        << QDateTime::currentDateTime().toString("yyyy-MM-dd HH:mm:ss") << "\n";
    out << (current ? "Симуляция: текущая\n" : "Симуляция: прошлая (путь был сброшен)\n");
    out << "Знаков после запятой в координатах: " << dec << "\n";
    out << "Углы направления луча: в градусах, из внутреннего представления луча, "
           "в пределах от 0 до 360\n\n";

    // Комната
    out << "КОМНАТА\n";
    out << "Сторон: " << snap.walls.size() << "\n";
    for (size_t i = 0; i < snap.walls.size(); ++i)
    {
        const auto &w = snap.walls[i];

        QString kind = "плоская";
        if (w.type == MirrorKind::Concave)
            kind = "вогнутая, R = " + num(w.radius);
        else if (w.type == MirrorKind::Convex)
            kind = "выпуклая, R = " + num(w.radius);

        out << "Точка " << (i + 1) << ": " << pt(w.x, w.y)
            << ", стена " << (i + 1) << " - " << kind << "\n";
    }

    // Начальные параметры луча
    out << "\nНАЧАЛЬНЫЙ ЛУЧ\n";
    out << "Источник: " << pt(hist[0].getX(), hist[0].getY())
        << " на стене " << (snap.beamWall + 1) << "\n";
    out << "Угол к стене: "
        << QString::number(snap.beamWallAngle, 'f', Style::SaveAngleDecimals) << "°\n";
    if (!states.empty())
        out << "Направление луча: " << ang(states[0].getA()) << "°\n";

    // Отражения: после каждой точки - направление луча, уходящего от неё
    out << "\nОТРАЖЕНИЯ\n";
    out << "Количество отражений: " << (hist.size() - 1) << "\n";
    for (size_t k = 1; k < hist.size(); ++k)
    {
        out << "Отражение " << k << ": " << pt(hist[k].getX(), hist[k].getY());
        if (k < states.size())
            out << ", угол луча после отражения: " << ang(states[k].getA()) << "°";
        out << "\n";
    }

    out.flush();

    if (out.status() != QTextStream::Ok || file.error() != QFileDevice::NoError)
    {
        AddMessage(
            QString("Ошибка при записи файла: %1 (%2).").arg(path, file.errorString()),
            MsgType::Error);
        file.close();
        return;
    }
    file.close();

    AddMessage(
        QString("История отражений (%1) сохранена в файл: %2")
            .arg(hist.size() - 1)
            .arg(path));
}

// Считает следующее отражение и запускает анимацию отрезка до него.
// Длительность - длина отрезка, делённая на скорость луча.
// false - шаг не получился (сообщение об ошибке уже в чате).
bool MainWindow::StartReflectionStep()
{
    // Сообщения о шаге (точка отражения, зацикливание) покажем в конце анимации
    pendingNotes_.clear();
    bufferNotes_ = true;

    if (!polygonCreator.NextReflection())
    {
        FlushPendingNotes();
        return false;
    }

    const auto &hist = polygonCreator.history_;
    const double length = hist[hist.size() - 2].Dist(hist.back());

    int ms = static_cast<int>(std::lround(length / raySpeed_ * Style::MsPerSecond));
    if (ms < Style::MinAnimMs)
        ms = Style::MinAnimMs;

    animating_ = true;
    animProgress_ = 0.0;
    reflectAnim_->setDuration(ms);
    reflectAnim_->start();
    UpdateButtons();   // на время анимации кнопки тускнеют
    update();
    return true;
}

// Сохраняет скорость луча и обновляет подпись. Новая скорость
// применяется к следующему отрезку (во время анимации ползунок заблокирован).
void MainWindow::SetRaySpeed(double speed)
{
    raySpeed_ = speed;
    ui->label->setText(QString("Скорость луча: %1 пикс/с").arg(speed, 0, 'f', 0));
}

// Сохраняет задержку между отражениями (мс) и обновляет подпись
void MainWindow::SetRepeatDelay(int ms)
{
    repeatDelayMs_ = ms;
    ui->label_2->setText(
        QString("Задержка луча между отражениями (в режиме авто): %1 сек")
            .arg(ms / Style::MsPerSecond, 0, 'f', Style::DelayDecimals));
}

// Пауза закончилась: следующее отражение (если повтор не остановили)
void MainWindow::OnDelayFinished()
{
    delaying_ = false;

    if (!repeating_)
        return;

    if (StartReflectionStep())
        return;   // анимация следующего отрезка уже идёт

    // шаг не получился (причина уже в чате)
    repeating_ = false;
    stopRequested_ = false;
    AddMessage("Повтор остановлен.");
    UpdateButtons();
    update();
}

    ui->horizontalSlider_2->setVisible(on);
    ui->label_2->setVisible(on);
    ui->checkBox->setVisible(on);
    ui->label_3->setVisible(on);

// Скорость и задержку нельзя менять во время анимации/повтора
// и пока выбирается источник луча
bool MainWindow::SettingsBlocked() const
{
    return animating_ || repeating_ || (!Redacting && Beaming);
}

// Поле «Макс. кол-во отражений» работает только при включённой галочке
// «Удалять старые отражения» (и когда остальные настройки не заблокированы)
bool MainWindow::TraceLimitBlocked() const
{
    return SettingsBlocked() || !eraseOld_;
}

// Применяет число из поля «Макс. кол-во отражений» (Enter или потеря фокуса).
// То же самое число - ничего не сбрасывается; новое - путь луча сбрасывается.
void MainWindow::CommitTraceLimit()
{
    const QString current = QString::number(traceLimit_);

    // Поле скрыто или заблокировано: введённое не применяем, возвращаем прежнее
    if (!ui->textEdit->isVisible() || TraceLimitBlocked())
    {
        if (ui->textEdit->toPlainText() != current)
            ui->textEdit->setPlainText(current);
        return;
    }

    const QString typed = ui->textEdit->toPlainText().trimmed();
    bool ok = false;
    const int value = typed.toInt(&ok);

    if (!ok || value < Style::TraceLimitMin || value > Style::TraceLimitMax)
    {
        AddMessage(
            QString("Максимальное количество отражений: нужно целое число от %1 до %2 "
                    "(введено «%3»). Оставлено прежнее значение: %4.")
                .arg(Style::TraceLimitMin)
                .arg(Style::TraceLimitMax)
                .arg(typed, current),
            MsgType::Error);
        ui->textEdit->setPlainText(current);
        return;
    }

    ui->textEdit->setPlainText(QString::number(value));   // например, «050» -> «50»

    if (value == traceLimit_)
        return;   // число то же самое - ничего не сбрасываем

    traceLimit_ = value;
    polygonCreator.ResetTrace();
    UpdateButtons();
    update();

    AddMessage(QString("Максимальное количество отражений: %1. Путь луча сброшен.")
                   .arg(traceLimit_));
}

// Enter и потеря фокуса применяют число в поле ввода. Пока настройки заблокированы
// (а для поля числа - ещё и пока выключена галочка), глотаем нажатия, колесо и клавиши
// и пишем в чат.
bool MainWindow::eventFilter(QObject *obj, QEvent *event)
{
    const bool isLimitField = (obj == ui->textEdit || obj == ui->textEdit->viewport());

    if (obj == ui->textEdit)
    {
        if (event->type() == QEvent::FocusOut)
        {
            CommitTraceLimit();
            return false;   // фокус уходит как обычно
        }

        if (!TraceLimitBlocked() && event->type() == QEvent::KeyPress)
        {
            const int key = static_cast<QKeyEvent *>(event)->key();
            if (key == Qt::Key_Return || key == Qt::Key_Enter)
            {
                ui->textEdit->clearFocus();   // потеря фокуса применит значение
                return true;                  // перевода строки в поле не будет
            }
        }
    }

    QString what;
    if (obj == ui->horizontalSlider)
        what = "скорость луча";
    else if (obj == ui->horizontalSlider_2)
        what = "задержку луча";
    else if (obj == ui->checkBox)
        what = "режим удаления старых отражений";
    else if (isLimitField)
        what = "максимальное количество отражений";

    const bool blocked = isLimitField ? TraceLimitBlocked() : SettingsBlocked();

    if (!what.isEmpty() && blocked)
    {
        switch (event->type())
        {
        case QEvent::MouseButtonPress:
        case QEvent::MouseButtonDblClick:
        case QEvent::Wheel:
        case QEvent::KeyPress:
            if (animating_ || repeating_)
                AddMessage(
                    QString("Нельзя менять %1, пока идёт анимация или повтор. "
                            "Дождитесь конца анимации или остановите повтор.").arg(what),
                    MsgType::Warning);
            else if (!Redacting && Beaming)
                AddMessage(
                    QString("Нельзя менять %1, пока не выбран источник на стене. "
                            "Кликните по стене, откуда должен выйти луч.").arg(what),
                    MsgType::Warning);
            else
                AddMessage(
                    QString("Нельзя менять %1, пока не включена галочка "
                            "«Удалять старые отражения».").arg(what),
                    MsgType::Warning);
            return true;
        default:
            break;
        }
    }
    return QMainWindow::eventFilter(obj, event);
}

// Если идёт анимация или повтор - сообщает об этом в чат,
// и вызывающий код должен выйти
bool MainWindow::IgnoreDuringAnimation()
{
    if (!animating_ && !repeating_)
        return false;

    AddMessage(
        "Нельзя редактировать, пока идёт симуляция: "
        "дождитесь конца анимации отражения или остановите повтор.",
        MsgType::Warning);
    return true;
}

void MainWindow::FinishReflection()
{
    animating_ = false;
    animProgress_ = 1.0;

    FlushPendingNotes();

    // Повтор: сразу следующее отражение, пока не просили остановиться
    // и луч не зациклился
    if (repeating_)
    {
        if (!stopRequested_ && !polygonCreator.looped_)
        {
            // Пауза между отражениями: кнопка 4 остаётся активной и её можно
            // нажать, чтобы остановить повтор прямо во время паузы
            if (repeatDelayMs_ > 0)
            {
                delaying_ = true;
                delayTimer_->start(repeatDelayMs_);
                UpdateButtons();
                update();
                return;
            }

            if (StartReflectionStep())
                return;   // анимация следующего отрезка уже идёт
        }

        const bool looped = polygonCreator.looped_;
        repeating_ = false;
        stopRequested_ = false;
        AddMessage(looped ? "Повтор остановлен: луч зациклился."
                          : "Повтор остановлен.");
    }

    UpdateButtons();   // если луч зациклился - кнопки 3 и 4 гаснут
    update();
}

void MainWindow::FlushPendingNotes()
{
    bufferNotes_ = false;

    for (const auto &note : pendingNotes_)
        AddMessage(note.second, note.first);

    pendingNotes_.clear();
}

void MainWindow::mousePressEvent(QMouseEvent *event)
{
    // Клик по окну убирает фокус с поля ввода (введённое число применится)
    if (ui->textEdit->hasFocus())
        ui->textEdit->clearFocus();

    if (IgnoreDuringAnimation())
        return;

    int x = event->position().x();
    int y = event->position().y();

    if (x > Limits::MaxX || y > Limits::MaxY)
    {
        AddMessage(
            QString("Клик вне рабочей области (допустимо: x от %1 до %2, y от %3 до %4).")
                .arg(Limits::MinX, 0, 'f', 0)
                .arg(Limits::MaxX, 0, 'f', 0)
                .arg(Limits::MinY, 0, 'f', 0)
                .arg(Limits::MaxY, 0, 'f', 0),
            MsgType::Warning);
        return;
    }
    // Клик по точке источника - изменить угол (в любом режиме)
    if (polygonCreator.prototype.isBeamSet_)
    {
        const Beam &b = polygonCreator.prototype.beam_;
        double bx = x - b.getX();
        double by = y - b.getY();

        if (bx * bx + by * by <= Limits::ClickRadius * Limits::ClickRadius)
        {
            BeamAngleDialog angleDialog(
                this,
                polygonCreator.prototype.beamWallAngle_
                );
            connect(&angleDialog, &BeamAngleDialog::invalidAngle, this,
                    [this](const QString &msg) { AddMessage(msg, MsgType::Error); });
            angleDialog.move(
                this->geometry().center()
                - angleDialog.rect().center()
                );

            if (angleDialog.exec() == QDialog::Accepted)
            {
                polygonCreator.prototype.SetBeamAngle(angleDialog.getAngle());
                polygonCreator.ResetTrace();
                UpdateButtons();   // новый угол - зацикленность сброшена
                update();

                AddMessage(QString("Угол луча изменён: %1° к стене %2.")
                               .arg(angleDialog.getAngle(), 0, 'f', BeamAngle::Decimals)
                               .arg(polygonCreator.prototype.beamWall_ + 1));
            }
            return;
        }
    }

    if (!Redacting && !Beaming)
    {
        AddMessage(
            "Сейчас идёт симуляция: точки ставить нельзя. "
            "«Переставить источник света» - выбрать другой луч, "
            "«Редактировать комнату» - править стены.",
            MsgType::Warning);
        return;
    }
    if (!Redacting && Beaming)
    {
        BeamPlaceResult res =
            polygonCreator.prototype.PlaceBeam(Points(x, y), Limits::BeamWallTolerance, Limits::MinBeamVertexDist);

        if (res == BeamPlaceResult::NoWall)
        {
            AddMessage(
                QString("Невозможно поставить луч: рядом нет стены. "
                        "Кликните точно по линии зеркала (не дальше %1 пикселей).")
                    .arg(static_cast<int>(Limits::BeamWallTolerance)),
                MsgType::Error);
            return;
        }

        if (res == BeamPlaceResult::TooClose)
        {
            AddMessage(
                QString("Луч нельзя ставить так близко к вершине "
                        "(ближе %1 пикселей). Выберите точку подальше.")
                    .arg(static_cast<int>(Limits::MinBeamVertexDist)),
                MsgType::Error);
            return;
        }

        repaint();   // показать точку, пока открыто окно выбора угла

        BeamAngleDialog angleDialog(this);
        connect(&angleDialog, &BeamAngleDialog::invalidAngle, this,
                [this](const QString &msg) { AddMessage(msg, MsgType::Error); });
        angleDialog.move(
            this->geometry().center()
            - angleDialog.rect().center()
            );

        if (angleDialog.exec() == QDialog::Accepted)
        {
            auto &poly = polygonCreator.prototype;

            poly.SetBeamAngle(angleDialog.getAngle());
            polygonCreator.ResetTrace();
            Beaming = false;
            UpdateButtons();

            AddMessage(QString("Источник луча: (%1; %2) на стене %3, угол к стене %4°.")
                           .arg(poly.beam_.getX(), 0, 'f', Style::MessageDecimals)
                           .arg(poly.beam_.getY(), 0, 'f', Style::MessageDecimals)
                           .arg(poly.beamWall_ + 1)
                           .arg(angleDialog.getAngle(), 0, 'f', BeamAngle::Decimals));
        }
        else
        {
            polygonCreator.prototype.isBeamSet_ = false;
            AddMessage("Выбор угла отменён: источник луча не поставлен.",
                       MsgType::Warning);
        }
        update();
        return;
    }
    const auto &temp = polygonCreator.prototype.sides_;

    // =========================================================
    // СУЩЕСТВУЮЩИЕ ТОЧКИ — НАЧАЛА ЗЕРКАЛ
    // =========================================================

    for (int i = 0; i < temp.size(); ++i)
    {
        double dx =
            x - temp[i]->getStart().getX();

        double dy =
            y - temp[i]->getStart().getY();

        // Попали точно в точку
        if (dx * dx + dy * dy <= Limits::ClickRadius * Limits::ClickRadius)
        {
            // -------------------------------------------------
            // ПОЛИГОН УЖЕ ЗАМКНУТ
            // Диалог выбора, но «Подключить» недоступно
            // -------------------------------------------------

            if (polygonCreator.prototype.isCycled_)
            {
                ChangeOrConnectDialog dialog(this, true, false);

                dialog.move(
                    this->geometry().center()
                    - dialog.rect().center()
                    );

                const int choice = dialog.exec();

                if (choice == QDialog::Rejected)
                {
                    AddMessage("Действие с точкой отменено.");
                    return;
                }

                if (choice == ChangeOrConnectDialog::DeleteCode)
                {
                    polygonCreator.DeletePoint(i);   // удалит вершину, если останется не меньше Limits::MinWalls
                    update();
                    return;
                }

                // Нажали "Изменить"
                ChangePointDialog changeDialog(
                    i + 1,
                    temp[i]->getStart(),
                    temp[i].get(),
                    false,
                    this
                    );

                changeDialog.move(
                    this->geometry().center()
                    - changeDialog.rect().center()
                    );

                if (changeDialog.exec() == QDialog::Accepted)
                {
                    polygonCreator.ChangeMirror(
                        i,
                        changeDialog.getData()
                        );

                    update();
                }

                return;
            }

            // -------------------------------------------------
            // ПОЛИГОН НЕ ЗАМКНУТ
            // Показываем ChangeOrConnectDialog
            // -------------------------------------------------

            ChangeOrConnectDialog dialog(this);

            dialog.move(
                this->geometry().center()
                - dialog.rect().center()
                );

            const int choice = dialog.exec();

            if (choice == QDialog::Rejected)
            {
                // Esc или закрытие окна - ничего не делаем
                AddMessage("Действие с точкой отменено.");
                return;
            }

            // Нажали "Удалить"
            if (choice == ChangeOrConnectDialog::DeleteCode)
            {
                polygonCreator.DeletePoint(i);
                update();
                return;
            }

            // Accepted - "Изменить", иначе (ConnectCode) - "Подключить"
            if (choice == QDialog::Accepted)
            {
                // Нажали "Изменить"

                ChangePointDialog changeDialog(
                    i + 1,
                    temp[i]->getStart(),
                    temp[i].get(),
                    false,
                    this
                    );

                changeDialog.move(
                    this->geometry().center()
                    - changeDialog.rect().center()
                    );

                if (changeDialog.exec() == QDialog::Accepted)
                {
                    polygonCreator.ChangeMirror(
                        i,
                        changeDialog.getData()
                        );

                    update();
                }

                return;
            }
            else
            {
                // Нажали "Подключить"

                // Последнее зеркало нельзя соединять
                // само с собой
                if (i == temp.size() - 1)
                {
                    AddMessage("Нельзя соединять зеркало с собой же!", MsgType::Error);

                    return;
                }

                polygonCreator.ConnectPoints(i);

                update();

                return;
            }
        }

        // Слишком близко к существующей точке
        if (dx * dx + dy * dy <= Limits::MinPointDist * Limits::MinPointDist)
        {
            AddMessage(QString("Нельзя поставить точку: она слишком близко к другой точке (ближе %1 пикселей).")
                           .arg(static_cast<int>(Limits::MinPointDist)),
                       MsgType::Error);

            return;
        }
    }

    // =========================================================
    // ПОСЛЕДНЯЯ ТОЧКА НЕЗАМКНУТОГО ПОЛИГОНА
    // =========================================================

    if (!temp.empty()
        && !polygonCreator.prototype.isCycled_)
    {
        double dx =
            x - temp.back()->getEnd().getX();

        double dy =
            y - temp.back()->getEnd().getY();

        // Попали в последнюю точку
        if (dx * dx + dy * dy <= Limits::ClickRadius * Limits::ClickRadius)
        {
            // Последнюю точку нельзя подключить: в диалоге доступны
            // только "Изменить" и "Удалить"
            const int lastIdx = static_cast<int>(temp.size());

            ChangeOrConnectDialog dialog(this, true, false);

            dialog.move(
                this->geometry().center()
                - dialog.rect().center()
                );

            const int choice = dialog.exec();

            if (choice == QDialog::Rejected)
            {
                AddMessage("Действие с точкой отменено.");
                return;
            }

            if (choice == ChangeOrConnectDialog::DeleteCode)
            {
                polygonCreator.DeletePoint(lastIdx);
                update();
                return;
            }

            // Нажали "Изменить"

            ChangePointDialog changeDialog(
                temp.size() + 1,
                temp.back()->getEnd(),
                temp.back().get(),
                true,
                this
                );

            changeDialog.move(
                this->geometry().center()
                - changeDialog.rect().center()
                );

            if (changeDialog.exec() == QDialog::Accepted)
            {
                polygonCreator.ChangeMirror(
                    temp.size(),
                    changeDialog.getData()
                    );

                update();
            }

            return;
        }

        // Слишком близко к последней точке
        if (dx * dx + dy * dy <= Limits::MinPointDist * Limits::MinPointDist)
        {
            AddMessage(QString("Нельзя поставить точку: она слишком близко к другой точке (ближе %1 пикселей).")
                           .arg(static_cast<int>(Limits::MinPointDist)),
                       MsgType::Error);

            return;
        }
    }

    // =========================================================
    // ЕСЛИ НЕ ПОПАЛИ НИ В ОДНУ ТОЧКУ
    // СОЗДАЁМ НОВУЮ
    // =========================================================

    if (polygonCreator.prototype.isCycled_)
    {
        AddMessage(
            "Комната замкнута: новые точки ставить нельзя. "
            "Нажмите «Откатить точку», чтобы разомкнуть комнату, "
            "или кликните по вершине, чтобы изменить её.",
            MsgType::Error);
        return;
    }

    // Если существует только первая точка
    if (temp.empty()
        && polygonCreator.prototype.isExist())
    {
        double dx =
            x - polygonCreator.prototype.first.getX();

        double dy =
            y - polygonCreator.prototype.first.getY();

        // Клик по единственной точке: доступно только удаление
        if (dx * dx + dy * dy <= Limits::ClickRadius * Limits::ClickRadius)
        {
            ChangeOrConnectDialog dialog(this, false, false);

            dialog.move(
                this->geometry().center()
                - dialog.rect().center()
                );

            if (dialog.exec() == ChangeOrConnectDialog::DeleteCode)
            {
                polygonCreator.DeletePoint(0);
                update();
            }
            else
            {
                AddMessage("Действие с точкой отменено.");
            }

            return;
        }

        if (dx * dx + dy * dy <= Limits::MinPointDist * Limits::MinPointDist)
        {
            AddMessage(QString("Нельзя поставить точку: она слишком близко к другой точке (ближе %1 пикселей).")
                           .arg(static_cast<int>(Limits::MinPointDist)),
                       MsgType::Error);

            return;
        }
    }

    polygonCreator.CreatePoint(x, y);

    update();
}

void MainWindow::paintEvent(QPaintEvent *event)
{
    Q_UNUSED(event);

    QPainter painter(this);
    painter.setRenderHint(QPainter::Antialiasing, true);

    // Тёмный фон окна и рабочей зоны (центральный виджет прозрачный, см. Darktheme.h)
    painter.fillRect(rect(), Theme::Window);
    const QRectF zone(Limits::MinX, Limits::MinY,
                      Limits::MaxX - Limits::MinX, Limits::MaxY - Limits::MinY);
    painter.fillRect(zone, Theme::Canvas);
    painter.setPen(QPen(Theme::CanvasBorder, 1));
    painter.setBrush(Qt::NoBrush);
    painter.drawRect(zone);

    QPen pen(Theme::Wall);
    pen.setWidth(Style::WallWidth);

    painter.setPen(pen);
    painter.setBrush(Theme::Wall);

    const auto &temp = polygonCreator.prototype.sides_;

    if (!polygonCreator.prototype.isExist())
        return;

    painter.drawEllipse(
        QPointF(polygonCreator.prototype.first.getX(),
                polygonCreator.prototype.first.getY()),
        Style::PointRadius,
        Style::PointRadius
        );

    if (!temp.empty())
    {
        painter.drawLine(
            polygonCreator.prototype.first.getX(),
            polygonCreator.prototype.first.getY(),
            temp[0]->getStart().getX(),
            temp[0]->getStart().getY()
            );
    }

    for (int i = 0;
         i < polygonCreator.prototype.sides_.size();
         i++)
    {
        painter.drawEllipse(
            QPointF(temp[i]->getEnd().getX(),
                    temp[i]->getEnd().getY()),
            Style::PointRadius,
            Style::PointRadius
            );

        temp[i]->Draw(painter, true);
    }

    // Путь луча по нажатиям "Следующее отражение"
    const auto &hist = polygonCreator.history_;
    if (polygonCreator.HasTrace())
    {
        QPen tracePen(Theme::Trace);   // жёлтый
        tracePen.setWidth(Style::TraceWidth);
        painter.setPen(tracePen);

        // Если включено удаление, остаются только последние traceLimit_ отрезков (вместе с растущим).
        // Старый отрезок пропадает сразу, без анимации, как только начинается новый.
        // Сама история history_ не обрезается: по ней ищется зацикливание.
        size_t firstSeg = 1;
        const size_t maxLines = eraseOld_ ? static_cast<size_t>(traceLimit_) : size_t(0);
        if (maxLines > 0 && hist.size() - 1 > maxLines)
            firstSeg = hist.size() - maxLines;

        for (size_t k = firstSeg; k < hist.size(); ++k)
        {
            QPointF from(hist[k - 1].getX(), hist[k - 1].getY());
            QPointF to(hist[k].getX(), hist[k].getY());

            // Последний отрезок во время анимации растёт от нуля до полной длины
            if (animating_ && k == hist.size() - 1)
                to = from + (to - from) * animProgress_;

            painter.drawLine(from, to);
        }
    }

    // Оранжевая точка подсвечивает последнее достигнутое отражение.
    // Когда луч долетает до следующего, прежняя точка исчезает
    // (во время анимации ещё горит та, от которой луч стартовал).
    if (polygonCreator.HasTrace())
    {
        QPen reflectPen(Theme::Reflect);
        reflectPen.setWidth(Style::ReflectPointWidth);
        reflectPen.setCapStyle(Qt::RoundCap);
        painter.setPen(reflectPen);

        const size_t reached = animating_ ? hist.size() - 2 : hist.size() - 1;

        // индекс 0 - источник луча, он красный и рисуется отдельно
        if (reached >= 1)
            painter.drawPoint(QPointF(hist[reached].getX(), hist[reached].getY()));
    }

    if (polygonCreator.prototype.isBeamSet_)
    {
        const Beam &b = polygonCreator.prototype.beam_;
        QPen beamPen(Theme::Source);
        beamPen.setWidth(Style::WallWidth);
        painter.setPen(beamPen);
        painter.setBrush(Theme::Source);
        painter.drawEllipse(QPointF(b.getX(), b.getY()),
                            Style::BeamRadius,
                            Style::BeamRadius);
    }
}

namespace {

double ArcTol(double r) { return 1e-6 + 1e-12 * r; }

bool PointsEqual(const Points& a, const Points& b) {
    return a.Dist(b) < POINT_EPS;
}
void AddUniquePoint(std::vector<Points>& pts, const Points& p) {
    for (const auto& e : pts) if (PointsEqual(e, p)) return;
    pts.push_back(p);
}

// ---------- отрезок / отрезок ----------
void IntersectSegments(const Points& a1, const Points& a2,
                       const Points& b1, const Points& b2,
                       std::vector<Points>& result) {
    double d1x = a2.getX() - a1.getX(), d1y = a2.getY() - a1.getY();
    double d2x = b2.getX() - b1.getX(), d2y = b2.getY() - b1.getY();
    double L1 = std::sqrt(d1x * d1x + d1y * d1y);
    double L2 = std::sqrt(d2x * d2x + d2y * d2y);
    if (L1 < POINT_EPS || L2 < POINT_EPS) return;

    double wx = b1.getX() - a1.getX(), wy = b1.getY() - a1.getY();
    double cross = d1x * d2y - d1y * d2x;

    if (std::abs(cross) / (L1 * L2) < PAR_EPS) {
        double distToLine = std::abs(wx * d1y - wy * d1x) / L1;
        if (distToLine > POINT_EPS) return;

        double L1sq = L1 * L1;
        double tb1 = (wx * d1x + wy * d1y) / L1sq;
        double tb2 = ((b2.getX() - a1.getX()) * d1x +
                      (b2.getY() - a1.getY()) * d1y) / L1sq;
        double tol = POINT_EPS / L1;

        double lo = std::max(0.0, std::min(tb1, tb2));
        double hi = std::min(1.0, std::max(tb1, tb2));
        if (lo > hi + tol) return;
        if (lo > hi) lo = hi = std::min(1.0, std::max(0.0, (lo + hi) / 2));

        AddUniquePoint(result, Points(a1.getX() + lo * d1x, a1.getY() + lo * d1y));
        if (hi - lo > tol)
            AddUniquePoint(result, Points(a1.getX() + hi * d1x, a1.getY() + hi * d1y));
        return;
    }

    double t = (wx * d2y - wy * d2x) / cross;
    double u = (wx * d1y - wy * d1x) / cross;
    double tolT = POINT_EPS / L1, tolU = POINT_EPS / L2;

    if (t >= -tolT && t <= 1.0 + tolT && u >= -tolU && u <= 1.0 + tolU) {
        t = std::min(1.0, std::max(0.0, t));
        AddUniquePoint(result, Points(a1.getX() + t * d1x, a1.getY() + t * d1y));
    }
}

// ---------- отрезок / дуга ----------
void IntersectSegmentArc(const Points& a1, const Points& a2,
                         const Mirror& arc,
                         std::vector<Points>& result) {
    if (arc.isFlat()) {
        IntersectSegments(a1, a2, arc.getStart(), arc.getEnd(), result);
        return;
    }

    double dx = a2.getX() - a1.getX(), dy = a2.getY() - a1.getY();
    double L = std::sqrt(dx * dx + dy * dy);
    if (L < POINT_EPS) return;
    double ux = dx / L, uy = dy / L;

    Points c = arc.CircleCenter();
    double r = std::abs(arc.getRad());

    double wx = c.getX() - a1.getX(), wy = c.getY() - a1.getY();
    double proj = wx * ux + wy * uy;
    double perp = std::abs(wx * uy - wy * ux);

    double tol = EPS * std::max(1.0, r);
    if (perp > r + tol) return;
    double h = (perp > r - tol) ? 0.0 : std::sqrt((r - perp) * (r + perp));

    auto check = [&](double s) {
        if (s < -POINT_EPS || s > L + POINT_EPS) return;
        Points p(a1.getX() + s * ux, a1.getY() + s * uy);
        if (arc.PointOnArc(p, c)) AddUniquePoint(result, p);
    };

    check(proj - h);
    if (h > 0) check(proj + h);
}

// ---------- дуга / дуга ----------
void IntersectArcs(const Mirror& m1, const Mirror& m2,
                   std::vector<Points>& result) {
    Points c1 = m1.CircleCenter();
    Points c2 = m2.CircleCenter();
    double r1 = std::abs(m1.getRad());
    double r2 = std::abs(m2.getRad());

    double dx = c2.getX() - c1.getX();
    double dy = c2.getY() - c1.getY();
    double d = std::sqrt(dx * dx + dy * dy);
    double tol = EPS * std::max(1.0, std::max(r1, r2));

    if (d < POINT_EPS) {
        if (std::abs(r1 - r2) < POINT_EPS) {
            if (m1.PointOnArc(m2.getStart(), c1)) AddUniquePoint(result, m2.getStart());
            if (m1.PointOnArc(m2.getEnd(),   c1)) AddUniquePoint(result, m2.getEnd());
            if (m2.PointOnArc(m1.getStart(), c2)) AddUniquePoint(result, m1.getStart());
            if (m2.PointOnArc(m1.getEnd(),   c2)) AddUniquePoint(result, m1.getEnd());
        }
        return;
    }

    if (d > r1 + r2 + tol) return;
    if (d < std::abs(r1 - r2) - tol) return;

    double a = (r1 * r1 - r2 * r2 + d * d) / (2.0 * d);
    double hSq = r1 * r1 - a * a;
    double h = hSq > 0 ? std::sqrt(hSq) : 0.0;

    double px = c1.getX() + a * dx / d;
    double py = c1.getY() + a * dy / d;

    Points p1(px + h * dy / d, py - h * dx / d);
    Points p2(px - h * dy / d, py + h * dx / d);

    if (m1.PointOnArc(p1, c1) && m2.PointOnArc(p1, c2)) AddUniquePoint(result, p1);
    if (m1.PointOnArc(p2, c1) && m2.PointOnArc(p2, c2)) AddUniquePoint(result, p2);
}
} // namespace

Mirror::Mirror(Points st, Points end, double rad)
    : start_(st), end_(end), rad_(rad)
{
    double d = st.Dist(end);
    if (!(d >= POINT_EPS))
        throw std::invalid_argument("Mirror: start and end coincide");
    double r = std::abs(rad);
    if (r >= EPS && r <= FLAT_R && r < d / 2.0 - POINT_EPS)
        throw std::invalid_argument("Mirror: |radius| is less than half of the chord");
}

Mirror::Mirror(Points st, Points end)
    : Mirror(st, end, 0.0)
{
}

double Mirror::getRad()    const { return rad_; }
double Mirror::getRadius() const { return std::abs(rad_); }
Points Mirror::getStart()  const { return start_; }
Points Mirror::getEnd()    const { return end_; }
double Mirror::Dist()      const { return start_.Dist(end_); }

bool Mirror::isFlat() const
{
    double r = std::abs(rad_);
    return !(r >= EPS) || r > FLAT_R;
}

/*
 * Центр окружности со стороной, заданной знаком rad:
 * rad > 0 -> слева от start -> end, rad < 0 -> справа.
 */
Points Mirror::CircleCenter() const
{
    return CircleCenter(rad_ >= 0);
}

/*
 * Центр окружности с явно заданной стороной.
 *
 * leftSide = true  -> центр слева от направления start -> end
 * leftSide = false -> справа.
 *
 * Для отрезка возвращается середина хорды.
 */
Points Mirror::CircleCenter(bool leftSide) const
{
    Points mid((start_.getX() + end_.getX()) / 2.0,
               (start_.getY() + end_.getY()) / 2.0);
    if (isFlat()) return mid;

    double r = std::abs(rad_);
    double d = Dist();
    double half = d / 2.0;
    double h = std::sqrt(std::max(0.0, (r - half) * (r + half)));

    // Левая нормаль к start -> end: (-dy, dx)
    double perpX = -(end_.getY() - start_.getY()) / d;
    double perpY =  (end_.getX() - start_.getX()) / d;
    double sign = leftSide ? 1.0 : -1.0;

    return Points(mid.getX() + sign * h * perpX,
                  mid.getY() + sign * h * perpY);
}

bool Mirror::flatPointOnArc(const Points& p) const
{
    double dx = end_.getX() - start_.getX();
    double dy = end_.getY() - start_.getY();
    double d = std::sqrt(dx * dx + dy * dy);
    double cross = (p.getX() - start_.getX()) * dy - (p.getY() - start_.getY()) * dx;
    if (std::abs(cross) / d > POINT_EPS) return false;
    double t = ((p.getX() - start_.getX()) * dx +
                (p.getY() - start_.getY()) * dy) / (d * d);
    double tol = POINT_EPS / d;
    return t >= -tol && t <= 1.0 + tol;
}

bool Mirror::arcPointOnArc(const Points& p, const Points& center) const
{
    double r = std::abs(rad_);
    if (std::abs(p.Dist(center) - r) > ArcTol(r)) return false;

    double dx = end_.getX() - start_.getX();
    double dy = end_.getY() - start_.getY();
    double d = std::sqrt(dx * dx + dy * dy);

    double sign = (rad_ >= 0 ? 1.0 : -1.0);
    double nx =  sign * dy / d;
    double ny = -sign * dx / d;
    double mx = (start_.getX() + end_.getX()) / 2.0;
    double my = (start_.getY() + end_.getY()) / 2.0;

    return (p.getX() - mx) * nx + (p.getY() - my) * ny >= -POINT_EPS;
}

bool Mirror::PointOnArc(const Points& p, const Points& center) const
{
    if (isFlat()) return flatPointOnArc(p);
    return arcPointOnArc(p, center);
}

Points Mirror::ClosestPoint(const Points& p, double& dist) const
{
    if (isFlat()) {
        double dx = end_.getX() - start_.getX();
        double dy = end_.getY() - start_.getY();
        double len2 = dx * dx + dy * dy;
        double t = ((p.getX() - start_.getX()) * dx +
                    (p.getY() - start_.getY()) * dy) / len2;
        t = std::clamp(t, 0.0, 1.0);
        Points q(start_.getX() + t * dx, start_.getY() + t * dy);
        dist = p.Dist(q);
        return q;
    }

    Points c = CircleCenter();
    double r = getRadius();
    double vx = p.getX() - c.getX();
    double vy = p.getY() - c.getY();
    double len = std::sqrt(vx * vx + vy * vy);

    if (len > EPS) {
        Points q(c.getX() + vx / len * r, c.getY() + vy / len * r);
        if (PointOnArc(q, c)) {
            dist = std::abs(len - r);
            return q;
        }
    }

    double ds = p.Dist(start_);
    double de = p.Dist(end_);
    if (ds <= de) { dist = ds; return start_; }
    dist = de;
    return end_;
}

std::vector<Points> Mirror::Intersections(const Mirror& other) const
{
    std::vector<Points> result;
    bool f1 = isFlat(), f2 = other.isFlat();

    if (f1 && f2) {
        IntersectSegments(start_, end_, other.getStart(), other.getEnd(), result);
    } else if (f1) {
        IntersectSegmentArc(start_, end_, other, result);
    } else if (f2) {
        IntersectSegmentArc(other.getStart(), other.getEnd(), *this, result);
    } else {
        IntersectArcs(*this, other, result);
    }
    return result;
}

/*
 * Рисование круглой дуги.
 *
 * centerLeft:
 *   true  -> центр окружности слева от start -> end
 *   false -> справа.
 */
static void drawCircularArc(
    QPainter& painter,
    const Mirror& mirror,
    bool centerLeft
    )
{
    const double x1 = mirror.getStart().getX();
    const double y1 = mirror.getStart().getY();
    const double x2 = mirror.getEnd().getX();
    const double y2 = mirror.getEnd().getY();

    // Отрезок (rad == 0 или огромный радиус)
    if (mirror.isFlat())
    {
        painter.drawLine(QPointF(x1, y1), QPointF(x2, y2));
        return;
    }

    const double pi = 3.14159265358979323846;

    const Points center = mirror.CircleCenter(centerLeft);
    const double cx = center.getX();
    const double cy = center.getY();
    const double radius = mirror.getRadius();

    const double startAngle = std::atan2(y1 - cy, x1 - cx);
    const double endAngle   = std::atan2(y2 - cy, x2 - cx);

    /*
     * Величина МЕНЬШЕЙ дуги, нормализованная в [-pi; pi].
     * Знак определяется стороной центра: при центре слева
     * обход start -> end идёт в положительном направлении
     * atan2, при центре справа - в отрицательном.
     */
    double delta = endAngle - startAngle;

    while (delta > pi)
        delta -= 2.0 * pi;

    while (delta < -pi)
        delta += 2.0 * pi;

    delta = std::abs(delta) * (centerLeft ? 1.0 : -1.0);

    /*
     * Дуга рисуется ломаной: точки считаются в double, а концы ставятся
     * ровно в вершины. delta уже содержит знак обхода (в тех же углах atan2,
     * что и startAngle), поэтому пересчитывать направление под Qt не нужно.
     */
    const double arcLength = std::abs(delta) * radius;
    int segments = static_cast<int>(std::ceil(arcLength / MirrorDraw::SegmentLength));
    segments = std::max(segments, MirrorDraw::MinSegments);
    segments = std::min(segments, MirrorDraw::MaxSegments);

    QPolygonF poly;
    poly.reserve(segments + 1);
    for (int k = 0; k <= segments; ++k)
    {
        const double a = startAngle + delta * static_cast<double>(k) / segments;
        poly << QPointF(cx + radius * std::cos(a), cy + radius * std::sin(a));
    }
    poly.first() = QPointF(x1, y1);
    poly.last()  = QPointF(x2, y2);

    painter.drawPolyline(poly);
}


/*
 * Обычное плоское зеркало.
 */
void PlaneMirror::Draw(
    QPainter& painter,
    bool insideOnLeft
    ) const
{
    Q_UNUSED(insideOnLeft);

    painter.drawLine(
        QPointF(start_.getX(), start_.getY()),
        QPointF(end_.getX(), end_.getY())
        );
}


/*
 * ВЫПУКЛОЕ зеркало: центр окружности внутри полигона.
 */
void ConvexMirror::Draw(
    QPainter& painter,
    bool insideOnLeft
    ) const
{
    drawCircularArc(painter, *this, insideOnLeft);
}


/*
 * ВОГНУТОЕ зеркало: центр окружности снаружи полигона,
 * дуга выгибается внутрь комнаты.
 */
void ConcaveMirror::Draw(
    QPainter& painter,
    bool insideOnLeft
    ) const
{
    drawCircularArc(painter, *this, !insideOnLeft);
}
#ifndef MIRROR_H
#define MIRROR_H

#include "points.h"

#include <vector>

class QPainter;

// Рисование дуг: дуга выводится ломаной (drawArc с целыми углами в 1/16 градуса
// при большом радиусе смещает концы дуги на десятки пикселей)
#include "constants.h"

// Зеркало: отрезок (rad == 0 или очень большой радиус) либо МЕНЬШАЯ дуга
// окружности радиуса |rad| между start и end (до полуокружности включительно).
// Знак rad выбирает сторону центра относительно хорды start->end:
//   rad > 0  - центр слева от хорды (выпуклое), rad < 0 - справа (вогнутое).
class Mirror
{
public:
    // Бросает std::invalid_argument при нулевой хорде или если |rad| < хорда/2.
    Mirror(Points st, Points end, double rad);
    Mirror(Points st, Points end);          // плоское (rad = 0)
    virtual ~Mirror() = default;

    // ---- Базовые геттеры ----
    double getRad() const;                  // радиус СО ЗНАКОМ (как в n2mirror)
    double getRadius() const;               // |радиус| (удобно для рисования)
    Points getStart() const;
    Points getEnd() const;
    double Dist() const;

    // ---- Геометрия ----
    virtual bool   isFlat() const;

    // Центр окружности со стороной, заданной знаком rad.
    virtual Points CircleCenter() const;
    // Центр окружности с явно заданной стороной
    // (leftSide = true -> слева от start -> end).
    Points CircleCenter(bool leftSide) const;

    virtual bool PointOnArc(const Points& p, const Points& center) const;

    Points ClosestPoint(const Points& p, double& dist) const;

    // ---- Пересечения ----
    virtual std::vector<Points> Intersections(const Mirror& other) const;

    // ---- Рисование ----
    virtual void Draw(
        QPainter& painter,
        bool insideOnLeft
        ) const = 0;

protected:

bool Close(const Points& a, const Points& b) {
    return a.Dist(b) < POINT_EPS;
}

// Пересечение луча (dir - единичный) с отрезком. Возвращает t > 0 или -1.
double RaySegmentIntersect(const Points& origin, const Points& dir,
                           const Points& a, const Points& b) {
    double dx = b.getX() - a.getX();
    double dy = b.getY() - a.getY();
    double len = std::sqrt(dx * dx + dy * dy);
    if (len < EPS) return -1.0;

    double cross = dir.getX() * dy - dir.getY() * dx;
    if (std::abs(cross) / len < EPS) return -1.0; // параллелен

    double ox = a.getX() - origin.getX();
    double oy = a.getY() - origin.getY();

    double t = (ox * dy - oy * dx) / cross;
    double s = (ox * dir.getY() - oy * dir.getX()) / cross;
    double tolS = POINT_EPS / len;

    if (t > TRACE_EPS && s >= -tolS && s <= 1.0 + tolS) return t;
    return -1.0;
}

// Пересечение луча (dir - единичный) с дугой. Возвращает t > 0 или -1.
double RayArcIntersect(const Points& origin, const Points& dir, const Mirror& m) {
    Points c = m.CircleCenter();
    double r = std::abs(m.getRad());

    double fx = origin.getX() - c.getX();
    double fy = origin.getY() - c.getY();

    double b = fx * dir.getX() + fy * dir.getY();          // половина линейного коэф.
    double cc = fx * fx + fy * fy - r * r;
    double disc = b * b - cc;
    if (disc < 0) {
        if (disc < -EPS * std::max(1.0, r * r)) return -1.0;
        disc = 0;
    }
    double sq = std::sqrt(disc);
    double t1 = -b - sq;
    double t2 = -b + sq;

    double bestT = -1.0;
    for (double t : {t1, t2}) {
        if (t > TRACE_EPS && (bestT < 0 || t < bestT)) {
            Points p(origin.getX() + t * dir.getX(), origin.getY() + t * dir.getY());
            if (m.PointOnArc(p, c)) bestT = t;
        }
    }
    return bestT;
}

double RayMirrorIntersect(const Points& origin, const Points& dir, const Mirror& m) {
    return m.isFlat() ? RaySegmentIntersect(origin, dir, m.getStart(), m.getEnd())
                      : RayArcIntersect(origin, dir, m);
}

// Единичная нормаль, направленная против падающего луча.
Points GetNormal(const Points& hit, const Points& dir, const Mirror& m) {
    double nx, ny;
    if (m.isFlat()) {
        double dx = m.getEnd().getX() - m.getStart().getX();
        double dy = m.getEnd().getY() - m.getStart().getY();
        double len = std::sqrt(dx * dx + dy * dy);
        if (len < EPS) return Points(0, 0);
        nx = -dy / len;
        ny =  dx / len;
    } else {
        Points c = m.CircleCenter();
        double dx = hit.getX() - c.getX();
        double dy = hit.getY() - c.getY();
        double len = std::sqrt(dx * dx + dy * dy);
        if (len < EPS) return Points(0, 0);
        nx = dx / len;
        ny = dy / len;
    }
    if (nx * dir.getX() + ny * dir.getY() > 0) { nx = -nx; ny = -ny; }
    return Points(nx, ny);
}

// d' = d - 2(d·n)n, с перенормировкой от накопления ошибки
Points ReflectDirection(const Points& dir, const Points& n) {
    double dot = dir.getX() * n.getX() + dir.getY() * n.getY();
    double rx = dir.getX() - 2 * dot * n.getX();
    double ry = dir.getY() - 2 * dot * n.getY();
    double len = std::sqrt(rx * rx + ry * ry);
    if (len < EPS) return dir;
    return Points(rx / len, ry / len);
}

BeamPlaceResult Polygon::PlaceBeam(const Points& click, double maxDist, double minVertexDist) {
    double best = maxDist;
    int bestIdx = -1;
    Points bestPt;

    for (int i = 0; i < Size(); ++i) {
        double d;
        Points q = sides_[i]->ClosestPoint(click, d);
        if (d <= best) {
            best = d;
            bestIdx = i;
            bestPt = q;
        }
    }
    if (bestIdx < 0) return BeamPlaceResult::NoWall;

    // Слишком близко к вершине
    for (const auto& s : sides_) {
        if (s->getStart().Dist(bestPt) <= minVertexDist ||
            s->getEnd().Dist(bestPt)   <= minVertexDist)
            return BeamPlaceResult::TooClose;
    }

    beam_ = Beam(bestPt.getX(), bestPt.getY(), beam_.getA());
    isBeamSet_ = true;
    beamWall_ = bestIdx;
    return BeamPlaceResult::Placed;
}

void Polygon::SetBeamAngle(double wallAngleDeg) {
    if (!isBeamSet_ || beamWall_ < 0 || beamWall_ >= Size()) return;

    beamWallAngle_ = wallAngleDeg;

    const Mirror& m = *sides_[beamWall_];
    const double pi = 3.14159265358979323846;

    // касательная в точке источника по ходу обхода start -> end
    double tx, ty;
    if (m.isFlat()) {
        double dx = m.getEnd().getX() - m.getStart().getX();
        double dy = m.getEnd().getY() - m.getStart().getY();
        double len = std::sqrt(dx * dx + dy * dy);
        tx = dx / len;
        ty = dy / len;
    } else {
        Points c = m.CircleCenter();
        double rx = beam_.getX() - c.getX();
        double ry = beam_.getY() - c.getY();
        double len = std::sqrt(rx * rx + ry * ry);
        rx /= len;
        ry /= len;
        double sg = (m.getRad() >= 0) ? 1.0 : -1.0;
        tx = -sg * ry;
        ty =  sg * rx;
    }

    // сторона комнаты по знаку площади контура
    double area = 0.0;
    for (const auto& s : sides_) {
        area += s->getStart().getX() * s->getEnd().getY()
                - s->getEnd().getX()   * s->getStart().getY();
    }
    double side = (area >= 0) ? 1.0 : -1.0;

    // нормаль к стене в сторону комнаты
    double nx = -side * ty;
    double ny =  side * tx;

    double th = wallAngleDeg * pi / 180.0;
    double dx = tx * std::cos(th) + nx * std::sin(th);
    double dy = ty * std::cos(th) + ny * std::sin(th);

    beam_ = Beam(beam_.getX(), beam_.getY(), std::atan2(dy, dx));
}

bool Polygon::Check() const {
    const int n = Size();
    if (n < 2) return false;

    // Вырожденный двуугольник из двух отрезков - не многоугольник
    if (n == 2 && sides_[0]->isFlat() && sides_[1]->isFlat()) return false;

    // Контур должен быть замкнутым и непрерывным
    for (int i = 0; i < n; ++i) {
        if (!Close(sides_[i]->getEnd(), sides_[(i + 1) % n]->getStart())) return false;
    }

    // Любая пара сторон: общие точки допустимы только в общих вершинах соседей
    for (int i = 0; i < n; ++i) {
        for (int j = i + 1; j < n; ++j) {
            std::vector<Points> allowed;
            if (j == i + 1)            allowed.push_back(sides_[i]->getEnd());
            if (i == 0 && j == n - 1)  allowed.push_back(sides_[j]->getEnd());

            for (const Points& p : sides_[i]->Intersections(*sides_[j])) {
                bool ok = false;
                for (const Points& a : allowed) {
                    if (Close(p, a)) { ok = true; break; }
                }
                if (!ok) return false;
            }
        }
    }
    return true;
}


std::pair<Beam, Points> Polygon::TraceBeamIter(const Beam& initial) const {
    Points pos(initial.getX(), initial.getY());

    Points dir(
        std::cos(initial.getA()),
        std::sin(initial.getA())
        );

    double bestT = -1.0;
    int bestIdx = -1;

    // Ищем ближайшее зеркало
    for (int i = 0; i < Size(); ++i) {
        double t = RayMirrorIntersect(pos, dir, *sides_[i]);

        if (t > TRACE_EPS && (bestT < 0 || t < bestT)) {
            bestT = t;
            bestIdx = i;
        }
    }
    // Луч никуда не попал: остаёмся на месте, а не обращаемся к sides_[-1]
    if (bestIdx < 0) return {initial, pos};
    // Точка отражения
    Points hit(
        pos.getX() + bestT * dir.getX(),
        pos.getY() + bestT * dir.getY()
        );

    // Рассчитываем отражённое направление
    Points normal = GetNormal(hit, dir, *sides_[bestIdx]);
    Points reflectedDir = ReflectDirection(dir, normal);

    double newAngle = std::atan2(
        reflectedDir.getY(),
        reflectedDir.getX()
        );

    // Создаём новый луч из точки отражения
    Beam reflectedBeam(
        hit.getX(),
        hit.getY(),
        newAngle
        );

    return {reflectedBeam, hit};
}
